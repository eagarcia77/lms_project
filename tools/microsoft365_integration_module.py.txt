def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _insert_id(conn: Any, sql: str, params: tuple[Any, ...]) -> int:
    if database_url().startswith("postgres"):
        row = execute(conn, sql + " RETURNING id", params).fetchone()
        return int(row[0])
    return int(execute(conn, sql, params).lastrowid)


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _config() -> dict[str, str]:
    return {
        "client_id": os.getenv("MICROSOFT_CLIENT_ID", "").strip(),
        "client_secret": os.getenv("MICROSOFT_CLIENT_SECRET", "").strip(),
        "tenant": os.getenv("MICROSOFT_TENANT_ID", "organizations").strip() or "organizations",
        "redirect_uri": os.getenv("MICROSOFT_REDIRECT_URI", "").strip(),
        "scopes": os.getenv("MICROSOFT_SCOPES", DEFAULT_SCOPES).strip() or DEFAULT_SCOPES,
    }


def _configured() -> bool:
    cfg = _config()
    return bool(cfg["client_id"] and cfg["client_secret"] and cfg["redirect_uri"])


def _authority_url(kind: str) -> str:
    tenant = _config()["tenant"]
    if not tenant.replace("-", "").replace(".", "").isalnum():
        raise HTTPException(500, "MICROSOFT_TENANT_ID contains unsupported characters.")
    return f"https://login.microsoftonline.com/{tenant}/oauth2/v2.0/{kind}"


def _fernet() -> Fernet:
    seed = os.getenv("MICROSOFT_TOKEN_ENCRYPTION_KEY", "").strip() or os.getenv("SESSION_SECRET", "").strip() or os.getenv("NEXUS_SESSION_SECRET", "").strip()
    if len(seed) < 16:
        raise RuntimeError("Configure MICROSOFT_TOKEN_ENCRYPTION_KEY or a strong session secret before storing Microsoft tokens.")
    key = base64.urlsafe_b64encode(hashlib.sha256(seed.encode("utf-8")).digest())
    return Fernet(key)


def _encrypt(value: str) -> str:
    return _fernet().encrypt(value.encode("utf-8")).decode("ascii") if value else ""


def _decrypt(value: str) -> str:
    if not value:
        return ""
    try:
        return _fernet().decrypt(value.encode("ascii")).decode("utf-8")
    except InvalidToken as exc:
        raise HTTPException(500, "Stored Microsoft connection cannot be decrypted with the current server key.") from exc


def _utc_dt(value: str) -> datetime:
    try:
        parsed = datetime.fromisoformat(str(value or "").strip().replace("Z", "+00:00"))
    except ValueError:
        return datetime.fromtimestamp(0, tz=timezone.utc)
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=timezone.utc)
    return parsed.astimezone(timezone.utc)


def ensure_microsoft365_schema() -> None:
    pk = _pk()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_microsoft_connections (
            id {pk}, user_email TEXT NOT NULL UNIQUE, tenant_id TEXT, account_id TEXT,
            display_name TEXT, access_token_enc TEXT NOT NULL, refresh_token_enc TEXT,
            expires_at TEXT NOT NULL, scopes TEXT, created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_microsoft_event_links (
            id {pk}, course_event_id INTEGER NOT NULL UNIQUE, microsoft_event_id TEXT NOT NULL,
            web_url TEXT, join_url TEXT, created_by TEXT NOT NULL, created_at TEXT NOT NULL
        )""")


def _connection(conn: Any, email: str) -> dict[str, Any] | None:
    found = rows(execute(conn, "SELECT * FROM nuvedra_microsoft_connections WHERE lower(user_email)=?", (email.lower(),)))
    return found[0] if found else None


def _store_connection(conn: Any, email: str, *, tenant_id: str, account_id: str, display_name: str, access_token: str, refresh_token: str, expires_in: int, scopes: str) -> None:
    now = utcnow()
    expires_at = (datetime.now(timezone.utc) + timedelta(seconds=max(60, int(expires_in or 3600)))).isoformat()
    existing = _connection(conn, email)
    refresh = refresh_token or (_decrypt(str(existing.get("refresh_token_enc") or "")) if existing else "")
    execute(conn, """INSERT INTO nuvedra_microsoft_connections
        (user_email,tenant_id,account_id,display_name,access_token_enc,refresh_token_enc,expires_at,scopes,created_at,updated_at)
        VALUES (?,?,?,?,?,?,?,?,?,?)
        ON CONFLICT(user_email) DO UPDATE SET tenant_id=excluded.tenant_id,account_id=excluded.account_id,
        display_name=excluded.display_name,access_token_enc=excluded.access_token_enc,
        refresh_token_enc=excluded.refresh_token_enc,expires_at=excluded.expires_at,
        scopes=excluded.scopes,updated_at=excluded.updated_at""",
        (email.lower(), tenant_id, account_id, display_name, _encrypt(access_token), _encrypt(refresh), expires_at, scopes, now, now))


def _academic_user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def _has_author_course(conn: Any, email: str) -> bool:
    role_list = ",".join("?" for _ in academic_access.AUTHOR_ROLES)
    found = rows(execute(conn, f"""SELECT id FROM nexus_admin_enrollments
        WHERE lower(user_email)=? AND status='active' AND course_role IN ({role_list}) LIMIT 1""",
        (email.lower(),) + tuple(academic_access.AUTHOR_ROLES)))
    return bool(found)


def _require_author_any(conn: Any, email: str) -> None:
    if not _has_author_course(conn, email):
        raise HTTPException(403, "Microsoft 365 faculty resources require an active course-author role.")


def _safe_next(value: str, default: str = "/portal") -> str:
    clean = str(value or "").strip()
    return clean if clean.startswith("/") and not clean.startswith("//") else default


def _safe_web_url(value: str) -> str:
    clean = str(value or "").strip()
    parsed = urlparse(clean)
    if parsed.scheme not in {"http", "https"} or not parsed.netloc:
        raise HTTPException(400, "A valid http/https Microsoft 365 resource URL is required.")
    return clean


def _office_type(name: str, mime_type: str = "") -> str:
    lower = str(name or "").lower()
    if lower.endswith((".doc", ".docx")) or "word" in mime_type:
        return "document"
    if lower.endswith((".ppt", ".pptx")) or "presentation" in mime_type or "powerpoint" in mime_type:
        return "presentation"
    if lower.endswith((".xls", ".xlsx", ".csv")) or "spreadsheet" in mime_type or "excel" in mime_type:
        return "spreadsheet"
    if lower.endswith(".pdf") or mime_type == "application/pdf":
        return "pdf"
    return "link"


async def _token_exchange(payload: dict[str, str]) -> dict[str, Any]:
    try:
        async with httpx.AsyncClient(timeout=15.0, follow_redirects=False) as client:
            response = await client.post(_authority_url("token"), data=payload, headers={"Accept": "application/json"})
    except httpx.HTTPError as exc:
        raise HTTPException(502, "Microsoft identity service is unavailable.") from exc
    try:
        body = response.json()
    except ValueError:
        body = {}
    if response.status_code >= 400 or not body.get("access_token"):
        detail = str(body.get("error_description") or body.get("error") or "Microsoft token exchange failed.")
        raise HTTPException(502, detail[:500])
    return body


async def _access_token(email: str, *, force_refresh: bool = False) -> str:
    if not _configured():
        raise HTTPException(503, "Microsoft 365 is not configured on this NUVEDRA deployment.")
    with db() as conn:
        connection = _connection(conn, email)
    if not connection:
        raise HTTPException(401, "Connect Microsoft 365 before using this feature.")
    if not force_refresh and _utc_dt(str(connection.get("expires_at") or "")) > datetime.now(timezone.utc) + timedelta(seconds=90):
        return _decrypt(str(connection.get("access_token_enc") or ""))
    refresh = _decrypt(str(connection.get("refresh_token_enc") or ""))
    if not refresh:
        raise HTTPException(401, "Microsoft connection expired. Connect Microsoft 365 again.")
    cfg = _config()
    token = await _token_exchange({"client_id": cfg["client_id"], "client_secret": cfg["client_secret"], "grant_type": "refresh_token", "refresh_token": refresh, "redirect_uri": cfg["redirect_uri"], "scope": cfg["scopes"]})
    with db() as conn:
        _store_connection(conn, email, tenant_id=str(connection.get("tenant_id") or cfg["tenant"]), account_id=str(connection.get("account_id") or ""), display_name=str(connection.get("display_name") or email), access_token=str(token["access_token"]), refresh_token=str(token.get("refresh_token") or refresh), expires_in=int(token.get("expires_in") or 3600), scopes=str(token.get("scope") or connection.get("scopes") or cfg["scopes"]))
    return str(token["access_token"])


async def _graph_json(email: str, method: str, path: str, *, params: dict[str, Any] | None = None, json_body: dict[str, Any] | None = None) -> dict[str, Any]:
    if not path.startswith("/"):
        raise HTTPException(500, "Microsoft Graph path must be relative.")
    token = await _access_token(email)
    url = GRAPH_ROOT + path
    try:
        async with httpx.AsyncClient(timeout=18.0, follow_redirects=False) as client:
            response = await client.request(method.upper(), url, params=params, json=json_body, headers={"Authorization": f"Bearer {token}", "Accept": "application/json"})
        if response.status_code == 401:
            token = await _access_token(email, force_refresh=True)
            async with httpx.AsyncClient(timeout=18.0, follow_redirects=False) as client:
                response = await client.request(method.upper(), url, params=params, json=json_body, headers={"Authorization": f"Bearer {token}", "Accept": "application/json"})
    except httpx.HTTPError as exc:
        raise HTTPException(502, "Microsoft Graph is unavailable.") from exc
    try:
        body = response.json() if response.content else {}
    except ValueError:
        body = {}
    if response.status_code >= 400:
        error = body.get("error") if isinstance(body, dict) else None
        message = error.get("message") if isinstance(error, dict) else None
        status = response.status_code if response.status_code in {400, 401, 403, 404, 409} else 502
        raise HTTPException(status, str(message or "Microsoft Graph request failed.")[:500])
    return body if isinstance(body, dict) else {}

def register_microsoft365_integration(app: FastAPI) -> None:
    ensure_microsoft365_schema()

    @app.get("/portal/microsoft-connect", response_model=None)
    async def microsoft_connect(request: Request, next: str = "/portal"):
        request.session["post_microsoft_redirect"] = _safe_next(next)
        request.session["microsoft_connect_only"] = bool(academic_access.google_user(request))
        return RedirectResponse("/auth/microsoft/login", status_code=303)

    @app.get("/auth/microsoft/login", response_model=None)
    async def microsoft_login(request: Request):
        if not _configured():
            raise HTTPException(503, "Microsoft 365 sign-in is not configured. Set MICROSOFT_CLIENT_ID, MICROSOFT_CLIENT_SECRET, MICROSOFT_TENANT_ID, and MICROSOFT_REDIRECT_URI.")
        cfg = _config()
        state = secrets.token_urlsafe(32)
        verifier = secrets.token_urlsafe(64)
        challenge = base64.urlsafe_b64encode(hashlib.sha256(verifier.encode("ascii")).digest()).decode("ascii").rstrip("=")
        request.session["microsoft_oauth_state"] = state
        request.session["microsoft_pkce_verifier"] = verifier
        params = {"client_id": cfg["client_id"], "response_type": "code", "redirect_uri": cfg["redirect_uri"], "response_mode": "query", "scope": cfg["scopes"], "state": state, "code_challenge": challenge, "code_challenge_method": "S256", "prompt": "select_account"}
        return RedirectResponse(_authority_url("authorize") + "?" + urlencode(params), status_code=303)

    @app.get("/auth/microsoft/callback", response_model=None)
    async def microsoft_callback(request: Request, code: str = "", state: str = "", error: str = "", error_description: str = ""):
        expected = str(request.session.pop("microsoft_oauth_state", "") or "")
        verifier = str(request.session.pop("microsoft_pkce_verifier", "") or "")
        if error:
            raise HTTPException(400, (error_description or error)[:500])
        if not code or not state or not expected or not secrets.compare_digest(state, expected) or not verifier:
            raise HTTPException(400, "Microsoft OAuth state validation failed.")
        cfg = _config()
        token = await _token_exchange({"client_id": cfg["client_id"], "client_secret": cfg["client_secret"], "grant_type": "authorization_code", "code": code, "redirect_uri": cfg["redirect_uri"], "code_verifier": verifier, "scope": cfg["scopes"]})
        access = str(token["access_token"])
        try:
            async with httpx.AsyncClient(timeout=15.0, follow_redirects=False) as client:
                me_response = await client.get(GRAPH_ROOT + "/me", headers={"Authorization": f"Bearer {access}", "Accept": "application/json"}, params={"$select": "id,displayName,mail,userPrincipalName"})
            me = me_response.json()
        except (httpx.HTTPError, ValueError) as exc:
            raise HTTPException(502, "Microsoft Graph could not validate the signed-in account.") from exc
        if me_response.status_code >= 400 or not isinstance(me, dict):
            raise HTTPException(502, "Microsoft Graph could not validate the signed-in account.")
        email = str(me.get("mail") or me.get("userPrincipalName") or "").strip().lower()
        if not email or "@" not in email:
            raise HTTPException(400, "Microsoft account did not provide a usable email address.")
        current = academic_access.google_user(request)
        connect_only = bool(request.session.pop("microsoft_connect_only", False))
        if connect_only and current and str(current.get("email") or "").lower() != email:
            raise HTTPException(409, "Connect the Microsoft account that uses the same email as your NUVEDRA academic identity.")
        with db() as conn:
            _store_connection(conn, email, tenant_id=cfg["tenant"], account_id=str(me.get("id") or ""), display_name=str(me.get("displayName") or email), access_token=access, refresh_token=str(token.get("refresh_token") or ""), expires_in=int(token.get("expires_in") or 3600), scopes=str(token.get("scope") or cfg["scopes"]))
            audit(conn, email, "microsoft365_connected", "user", email, cfg["tenant"], request.client.host if request.client else "")
        if not current:
            request.session["user"] = {"id": str(me.get("id") or email), "email": email, "name": str(me.get("displayName") or email), "_auth_source": "microsoft"}
        target = _safe_next(str(request.session.pop("post_microsoft_redirect", "/portal") or "/portal"))
        return RedirectResponse(target, status_code=303)

    @app.post("/portal/microsoft-disconnect", response_model=None)
    async def microsoft_disconnect(request: Request):
        user = _academic_user(request, "/portal")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            execute(conn, "DELETE FROM nuvedra_microsoft_connections WHERE lower(user_email)=?", (user["email"].lower(),))
            audit(conn, user["email"], "microsoft365_disconnected", "user", user["email"], "", request.client.host if request.client else "")
        return RedirectResponse("/portal", status_code=303)
