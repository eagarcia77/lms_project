from __future__ import annotations

from typing import Any

from fastapi import FastAPI, Form, HTTPException, Request
from fastapi.responses import HTMLResponse, RedirectResponse

import app.academic_access as academic_access
from app.admin_console import audit, database_url, db, execute, rows, utcnow

STUDIO_PREFIX = "/faculty/studio"
GROUP_ACTIVITY_TYPES = {"assignment", "project", "presentation"}


def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def _insert_id(conn: Any, sql: str, params: tuple[Any, ...]) -> int:
    if database_url().startswith("postgres"):
        row = execute(conn, sql + " RETURNING id", params).fetchone()
        return int(row[0])
    cursor = execute(conn, sql, params)
    return int(cursor.lastrowid)


def ensure_people_groups_schema() -> None:
    pk = _pk()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_groups (
            id {pk}, course_id INTEGER NOT NULL, name TEXT NOT NULL, description TEXT,
            status TEXT NOT NULL DEFAULT 'active', created_by TEXT NOT NULL,
            created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_group_members (
            id {pk}, group_id INTEGER NOT NULL, user_email TEXT NOT NULL,
            member_role TEXT NOT NULL DEFAULT 'member', created_at TEXT NOT NULL,
            UNIQUE(group_id,user_email)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_group_item_links (
            id {pk}, group_id INTEGER NOT NULL, item_id INTEGER NOT NULL,
            created_by TEXT NOT NULL, created_at TEXT NOT NULL,
            UNIQUE(group_id,item_id)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_group_posts (
            id {pk}, group_id INTEGER NOT NULL, author_email TEXT NOT NULL,
            body TEXT NOT NULL, parent_id INTEGER, created_at TEXT NOT NULL,
            updated_at TEXT NOT NULL
        )""")
        # Keep this module safe if Calendar & Notifications is ever disabled in a custom build.
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_notifications (
            id {pk}, recipient_email TEXT NOT NULL, course_id INTEGER, title TEXT NOT NULL,
            message TEXT, href TEXT, kind TEXT NOT NULL DEFAULT 'course_event',
            read_at TEXT, created_at TEXT NOT NULL
        )""")


def _group(conn: Any, group_id: int) -> dict[str, Any]:
    found = rows(execute(conn, """SELECT g.*,c.course_code,c.title AS course_title,c.status AS course_status
        FROM nuvedra_groups g JOIN nexus_admin_courses c ON c.id=g.course_id
        WHERE g.id=?""", (group_id,)))
    if not found:
        raise HTTPException(404, "Group not found.")
    return found[0]


def _group_members(conn: Any, group_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT gm.*,e.course_role,e.status AS enrollment_status
        FROM nuvedra_group_members gm
        JOIN nuvedra_groups g ON g.id=gm.group_id
        LEFT JOIN nexus_admin_enrollments e
          ON e.course_id=g.course_id AND lower(e.user_email)=lower(gm.user_email)
        WHERE gm.group_id=? ORDER BY gm.member_role DESC,lower(gm.user_email)""", (group_id,)))


def _course_people(conn: Any, course_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT id,user_email,course_role,status,created_at
        FROM nexus_admin_enrollments WHERE course_id=?
        ORDER BY CASE course_role
          WHEN 'instructor' THEN 1 WHEN 'teaching_assistant' THEN 2
          WHEN 'course_builder' THEN 3 WHEN 'facilitator' THEN 4
          WHEN 'student' THEN 5 WHEN 'observer' THEN 6 ELSE 7 END,
          lower(user_email)""", (course_id,)))


def _course_groups(conn: Any, course_id: int) -> list[dict[str, Any]]:
    groups = rows(execute(conn, """SELECT * FROM nuvedra_groups
        WHERE course_id=? AND status='active' ORDER BY name,id""", (course_id,)))
    for group in groups:
        group["member_count"] = int(rows(execute(conn,
            "SELECT COUNT(*) AS total FROM nuvedra_group_members WHERE group_id=?",
            (int(group["id"]),)))[0]["total"])
        group["activity_count"] = int(rows(execute(conn,
            "SELECT COUNT(*) AS total FROM nuvedra_group_item_links WHERE group_id=?",
            (int(group["id"]),)))[0]["total"])
    return groups


def _author_group(conn: Any, group_id: int, email: str) -> dict[str, Any]:
    group = _group(conn, group_id)
    academic_access.require_course_role(conn, int(group["course_id"]), email, academic_access.AUTHOR_ROLES)
    return group


def _student_group(conn: Any, group_id: int, email: str) -> tuple[dict[str, Any], dict[str, Any]]:
    group = _group(conn, group_id)
    if str(group.get("status")) != "active" or str(group.get("course_status")) != "active":
        raise HTTPException(403, "This group is not available.")
    academic_access.require_course_role(conn, int(group["course_id"]), email, {"student"})
    membership = rows(execute(conn, """SELECT * FROM nuvedra_group_members
        WHERE group_id=? AND lower(user_email)=?""", (group_id, email.lower())))
    if not membership:
        raise HTTPException(403, "You are not a member of this group.")
    return group, membership[0]


def group_context_for_item(conn: Any, item_id: int, email: str) -> dict[str, Any] | None:
    linked = rows(execute(conn, """SELECT g.id,g.name,g.course_id
        FROM nuvedra_group_item_links l JOIN nuvedra_groups g ON g.id=l.group_id
        WHERE l.item_id=? AND g.status='active' ORDER BY g.name,g.id""", (item_id,)))
    if not linked:
        return None
    group_ids = [int(row["id"]) for row in linked]
    placeholders = ",".join("?" for _ in group_ids)
    membership = rows(execute(conn, f"""SELECT g.id,g.name,g.course_id
        FROM nuvedra_group_members gm JOIN nuvedra_groups g ON g.id=gm.group_id
        WHERE gm.group_id IN ({placeholders}) AND lower(gm.user_email)=?
          AND g.status='active' ORDER BY g.name,g.id""", tuple(group_ids) + (email.lower(),)))
    return membership[0] if membership else {}


def require_group_item_access(conn: Any, item_id: int, email: str) -> dict[str, Any] | None:
    context = group_context_for_item(conn, item_id, email)
    if context == {}:
        raise HTTPException(403, "This activity is assigned to a different course group.")
    return context


def _notify(conn: Any, recipient: str, course_id: int, title: str, message: str, href: str, kind: str = "group") -> None:
    execute(conn, """INSERT INTO nuvedra_notifications
        (recipient_email,course_id,title,message,href,kind,read_at,created_at)
        VALUES (?,?,?,?,?,?,NULL,?)""",
        (recipient.lower(), course_id, title, message, href, kind, utcnow()))


def _notify_group(conn: Any, group_id: int, course_id: int, *, title: str, message: str,
                  href: str, exclude: str = "") -> int:
    members = rows(execute(conn, """SELECT lower(user_email) AS email
        FROM nuvedra_group_members WHERE group_id=? ORDER BY lower(user_email)""", (group_id,)))
    count = 0
    for member in members:
        email = str(member.get("email") or "").lower()
        if email and email != exclude.lower():
            _notify(conn, email, course_id, title, message, href, "group")
            count += 1
    return count


def _group_activities(conn: Any, group_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT l.id AS link_id,i.id AS item_id,i.title,i.item_type,i.status,
        i.points,i.due_at,m.title AS module_title,m.status AS module_status
        FROM nuvedra_group_item_links l
        JOIN nexus_content_items i ON i.id=l.item_id
        JOIN nexus_modules m ON m.id=i.module_id
        WHERE l.group_id=? ORDER BY m.position,m.id,i.position,i.id""", (group_id,)))


def _course_activity_options(conn: Any, course_id: int) -> list[dict[str, Any]]:
    placeholders = ",".join("?" for _ in GROUP_ACTIVITY_TYPES)
    return rows(execute(conn, f"""SELECT i.id,i.title,i.item_type,i.status,m.title AS module_title
        FROM nexus_content_items i JOIN nexus_modules m ON m.id=i.module_id
        WHERE m.course_id=? AND i.item_type IN ({placeholders})
        ORDER BY m.position,m.id,i.position,i.id""",
        (course_id,) + tuple(sorted(GROUP_ACTIVITY_TYPES))))

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/groups", response_model=None)
    async def create_group(course_id: int, request: Request, name: str = Form(...), description: str = Form("")):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/people")
        if isinstance(user, RedirectResponse):
            return user
        clean = name.strip()
        if not clean:
            raise HTTPException(400, "Group name is required.")
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            duplicate = rows(execute(conn, """SELECT id FROM nuvedra_groups
                WHERE course_id=? AND lower(name)=? AND status='active'""", (course_id, clean.lower())))
            if duplicate:
                raise HTTPException(409, "An active group with this name already exists.")
            now = utcnow()
            group_id = _insert_id(conn, """INSERT INTO nuvedra_groups
                (course_id,name,description,status,created_by,created_at,updated_at)
                VALUES (?,?,?,'active',?,?,?)""",
                (course_id, clean, description.strip(), user["email"], now, now))
            audit(conn, user["email"], "group_created", "group", str(group_id), clean,
                  request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/groups/{group_id}", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/groups/{{group_id}}/members", response_model=None)
    async def add_group_member(group_id: int, request: Request, student_email: str = Form(...), member_role: str = Form("member")):
        user = _user(request, f"{STUDIO_PREFIX}/groups/{group_id}")
        if isinstance(user, RedirectResponse):
            return user
        role = member_role if member_role in {"member", "leader"} else "member"
        email = student_email.strip().lower()
        with db() as conn:
            group = _author_group(conn, group_id, user["email"])
            course_id = int(group["course_id"])
            academic_access.require_course_role(conn, course_id, email, {"student"})
            existing = rows(execute(conn, """SELECT id FROM nuvedra_group_members
                WHERE group_id=? AND lower(user_email)=?""", (group_id, email)))
            if existing:
                execute(conn, "UPDATE nuvedra_group_members SET member_role=? WHERE id=?", (role, int(existing[0]["id"])))
            else:
                execute(conn, """INSERT INTO nuvedra_group_members
                    (group_id,user_email,member_role,created_at) VALUES (?,?,?,?)""",
                    (group_id, email, role, utcnow()))
            _notify(conn, email, course_id, f"Added to group: {group.get('name')}",
                    "You were added to a course collaboration group.", f"/learn/groups/{group_id}", "group")
            audit(conn, user["email"], "group_member_added", "group", str(group_id), email,
                  request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/groups/{group_id}", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/groups/{{group_id}}/members/{{member_id}}/remove", response_model=None)
    async def remove_group_member(group_id: int, member_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/groups/{group_id}")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            _author_group(conn, group_id, user["email"])
            member = rows(execute(conn, "SELECT * FROM nuvedra_group_members WHERE id=? AND group_id=?", (member_id, group_id)))
            if not member:
                raise HTTPException(404, "Group member not found.")
            execute(conn, "DELETE FROM nuvedra_group_members WHERE id=?", (member_id,))
            audit(conn, user["email"], "group_member_removed", "group", str(group_id),
                  str(member[0].get("user_email") or ""), request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/groups/{group_id}", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/groups/{{group_id}}/activities", response_model=None)
    async def assign_group_activity(group_id: int, request: Request, item_id: int = Form(...)):
        user = _user(request, f"{STUDIO_PREFIX}/groups/{group_id}")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            group = _author_group(conn, group_id, user["email"])
            course_id = int(group["course_id"])
            item_course_id, item, _module = academic_access.item_bundle(conn, item_id)
            if item_course_id != course_id:
                raise HTTPException(400, "The activity must belong to the same course as the group.")
            if str(item.get("item_type") or "") not in GROUP_ACTIVITY_TYPES:
                raise HTTPException(400, "This activity type cannot be assigned to a group.")
            duplicate = rows(execute(conn, "SELECT id FROM nuvedra_group_item_links WHERE group_id=? AND item_id=?", (group_id, item_id)))
            if not duplicate:
                execute(conn, """INSERT INTO nuvedra_group_item_links
                    (group_id,item_id,created_by,created_at) VALUES (?,?,?,?)""",
                    (group_id, item_id, user["email"], utcnow()))
            _notify_group(conn, group_id, course_id, title=f"Group activity: {item.get('title')}",
                          message=f"{group.get('name')} received a course activity.",
                          href=f"/learn/groups/{group_id}")
            audit(conn, user["email"], "group_activity_assigned", "item", str(item_id),
                  f"group={group_id}", request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/groups/{group_id}", status_code=303)

    @app.post("/learn/groups/{group_id}/posts", response_model=None)
    async def student_group_post(group_id: int, request: Request, body: str = Form(...)):
        user = _user(request, f"/learn/groups/{group_id}")
        if isinstance(user, RedirectResponse):
            return user
        clean = body.strip()
        if not clean:
            raise HTTPException(400, "Group message cannot be empty.")
        if len(clean) > 4000:
            raise HTTPException(400, "Group message is too long.")
        with db() as conn:
            group, _membership = _student_group(conn, group_id, user["email"])
            now = utcnow()
            _insert_id(conn, """INSERT INTO nuvedra_group_posts
                (group_id,author_email,body,parent_id,created_at,updated_at)
                VALUES (?,?,?,NULL,?,?)""", (group_id, user["email"], clean, now, now))
            _notify_group(conn, group_id, int(group["course_id"]),
                          title=f"New message in {group.get('name')}",
                          message=f"{user['email']} posted to your group.",
                          href=f"/learn/groups/{group_id}", exclude=user["email"])
            audit(conn, user["email"], "group_post_created", "group", str(group_id), "",
                  request.client.host if request.client else "")
        return RedirectResponse(f"/learn/groups/{group_id}", status_code=303)
