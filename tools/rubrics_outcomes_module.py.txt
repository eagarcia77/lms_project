from __future__ import annotations

from typing import Any

from fastapi import FastAPI, Form, HTTPException, Request
from fastapi.responses import HTMLResponse, RedirectResponse

import app.academic_access as academic_access
from app.admin_console import audit, database_url, db, execute, rows, utcnow

STUDIO_PREFIX = "/faculty/studio"


def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def ensure_rubrics_outcomes_schema() -> None:
    pk = _pk()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_rubrics (
            id {pk}, course_id INTEGER NOT NULL, title TEXT NOT NULL, description TEXT,
            status TEXT NOT NULL DEFAULT 'active', created_by TEXT NOT NULL,
            created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_rubric_criteria (
            id {pk}, rubric_id INTEGER NOT NULL, title TEXT NOT NULL, description TEXT,
            position INTEGER NOT NULL DEFAULT 1, created_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_rubric_levels (
            id {pk}, criterion_id INTEGER NOT NULL, label TEXT NOT NULL, description TEXT,
            points REAL NOT NULL DEFAULT 0, position INTEGER NOT NULL DEFAULT 1,
            created_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_rubric_links (
            id {pk}, rubric_id INTEGER NOT NULL, item_id INTEGER NOT NULL,
            created_by TEXT NOT NULL, created_at TEXT NOT NULL, UNIQUE(item_id)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_outcomes (
            id {pk}, course_id INTEGER NOT NULL, code TEXT NOT NULL, title TEXT NOT NULL,
            description TEXT, status TEXT NOT NULL DEFAULT 'active', created_by TEXT NOT NULL,
            created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_outcome_links (
            id {pk}, outcome_id INTEGER NOT NULL, item_id INTEGER NOT NULL,
            created_by TEXT NOT NULL, created_at TEXT NOT NULL, UNIQUE(outcome_id,item_id)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_rubric_evaluations (
            id {pk}, submission_id INTEGER NOT NULL, rubric_id INTEGER NOT NULL,
            raw_score REAL NOT NULL DEFAULT 0, possible_raw_score REAL NOT NULL DEFAULT 0,
            grade_points REAL NOT NULL DEFAULT 0, feedback TEXT,
            status TEXT NOT NULL DEFAULT 'graded', graded_by TEXT NOT NULL,
            graded_at TEXT NOT NULL, updated_at TEXT NOT NULL,
            UNIQUE(submission_id,rubric_id)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_rubric_scores (
            id {pk}, evaluation_id INTEGER NOT NULL, criterion_id INTEGER NOT NULL,
            level_id INTEGER NOT NULL, points_awarded REAL NOT NULL DEFAULT 0,
            feedback TEXT, UNIQUE(evaluation_id,criterion_id)
        )""")


def _course_items(conn: Any, course_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT i.id,i.title,i.item_type,i.points,i.status,m.title AS module_title,m.position AS module_position
        FROM nexus_content_items i JOIN nexus_modules m ON m.id=i.module_id
        WHERE m.course_id=? ORDER BY m.position,m.id,i.position,i.id""", (course_id,)))


def _rubric(conn: Any, rubric_id: int) -> dict[str, Any]:
    found = rows(execute(conn, "SELECT * FROM nuvedra_rubrics WHERE id=?", (rubric_id,)))
    if not found:
        raise HTTPException(404, "Rubric not found.")
    return found[0]


def _criteria(conn: Any, rubric_id: int) -> list[dict[str, Any]]:
    criteria = rows(execute(conn, "SELECT * FROM nuvedra_rubric_criteria WHERE rubric_id=? ORDER BY position,id", (rubric_id,)))
    for criterion in criteria:
        criterion["levels"] = rows(execute(conn, "SELECT * FROM nuvedra_rubric_levels WHERE criterion_id=? ORDER BY points DESC,position,id", (int(criterion["id"]),)))
    return criteria


def _rubric_max(criteria: list[dict[str, Any]]) -> float:
    return sum(max((float(level.get("points") or 0) for level in criterion.get("levels", [])), default=0.0) for criterion in criteria)


def _linked_rubric(conn: Any, item_id: int) -> dict[str, Any] | None:
    found = rows(execute(conn, """SELECT r.* FROM nuvedra_rubric_links l
        JOIN nuvedra_rubrics r ON r.id=l.rubric_id
        WHERE l.item_id=? AND r.status='active'""", (item_id,)))
    return found[0] if found else None


def _submission_bundle(conn: Any, submission_id: int) -> dict[str, Any]:
    found = rows(execute(conn, """SELECT s.*,i.title AS item_title,i.points AS item_points,i.id AS item_id,
        m.course_id,m.title AS module_title,c.course_code,c.title AS course_title
        FROM nuvedra_submissions s
        JOIN nexus_content_items i ON i.id=s.item_id
        JOIN nexus_modules m ON m.id=i.module_id
        JOIN nexus_admin_courses c ON c.id=m.course_id
        WHERE s.id=?""", (submission_id,)))
    if not found:
        raise HTTPException(404, "Submission not found.")
    return found[0]


def _outcome_attainment(conn: Any, outcome_id: int) -> tuple[float | None, int]:
    data = rows(execute(conn, """SELECT i.points,g.points_awarded FROM nuvedra_outcome_links ol
        JOIN nexus_content_items i ON i.id=ol.item_id
        JOIN nuvedra_submissions s ON s.item_id=i.id AND s.status='submitted'
        JOIN nuvedra_grades g ON g.submission_id=s.id AND g.status='graded'
        WHERE ol.outcome_id=? AND i.points IS NOT NULL AND i.points>0 AND g.points_awarded IS NOT NULL""", (outcome_id,)))
    percentages = [max(0.0, min(100.0, (float(row.get("points_awarded") or 0) / float(row.get("points") or 1)) * 100.0)) for row in data]
    if not percentages:
        return None, 0
    return sum(percentages) / len(percentages), len(percentages)


def _rubric_student_html(criteria: list[dict[str, Any]], selected: dict[int, dict[str, Any]] | None = None) -> str:
    cards = []
    selected = selected or {}
    for criterion in criteria:
        score = selected.get(int(criterion["id"]))
        levels = "".join(
            f'''<tr><td>{academic_access.esc(level.get('label'))}</td><td>{academic_access.esc(level.get('description'))}</td><td>{float(level.get('points') or 0):g}</td></tr>'''
            for level in criterion.get("levels", [])
        ) or '<tr><td colspan="3">No performance levels yet.</td></tr>'
        evaluated = ""
        if score:
            evaluated = f'''<p class="studio-notice"><strong data-i18n-en="Selected level" data-i18n-es="Nivel seleccionado">Selected level</strong>: {academic_access.esc(score.get('level_label'))} · {float(score.get('points_awarded') or 0):g} pts<br>{academic_access.esc(score.get('feedback') or '')}</p>'''
        cards.append(f'''<article class="studio-panel"><h3>{academic_access.esc(criterion.get('title'))}</h3><p>{academic_access.esc(criterion.get('description'))}</p>{evaluated}<div style="overflow:auto"><table><thead><tr><th data-i18n-en="Level" data-i18n-es="Nivel">Level</th><th data-i18n-en="Description" data-i18n-es="Descripción">Description</th><th data-i18n-en="Points" data-i18n-es="Puntos">Points</th></tr></thead><tbody>{levels}</tbody></table></div></article>''')
    return "".join(cards) or '<section class="studio-empty" data-i18n-en="This rubric has no criteria yet." data-i18n-es="Esta rúbrica todavía no tiene criterios.">This rubric has no criteria yet.</section>'

    @app.post(f"{STUDIO_PREFIX}/rubrics/{{rubric_id}}/criteria", response_model=None)
    async def add_criterion(rubric_id: int, request: Request, title: str = Form(...), description: str = Form("")):
        user = _user(request, f"{STUDIO_PREFIX}/rubrics/{rubric_id}")
        if isinstance(user, RedirectResponse): return user
        clean = title.strip()
        if not clean: raise HTTPException(400, "Criterion title is required.")
        with db() as conn:
            rubric = _rubric(conn, rubric_id)
            academic_access.require_course_role(conn, int(rubric["course_id"]), user["email"], academic_access.AUTHOR_ROLES)
            current = rows(execute(conn, "SELECT COUNT(*) AS total FROM nuvedra_rubric_criteria WHERE rubric_id=?", (rubric_id,)))
            execute(conn, "INSERT INTO nuvedra_rubric_criteria (rubric_id,title,description,position,created_at) VALUES (?,?,?,?,?)", (rubric_id, clean, description.strip(), int(current[0].get("total") or 0)+1, utcnow()))
        return RedirectResponse(f"{STUDIO_PREFIX}/rubrics/{rubric_id}", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/rubrics/{{rubric_id}}/criteria/{{criterion_id}}/levels", response_model=None)
    async def add_level(rubric_id: int, criterion_id: int, request: Request, label: str = Form(...), points: float = Form(...), description: str = Form("")):
        user = _user(request, f"{STUDIO_PREFIX}/rubrics/{rubric_id}")
        if isinstance(user, RedirectResponse): return user
        if points < 0: raise HTTPException(400, "Rubric points cannot be negative.")
        with db() as conn:
            rubric = _rubric(conn, rubric_id)
            academic_access.require_course_role(conn, int(rubric["course_id"]), user["email"], academic_access.AUTHOR_ROLES)
            found = rows(execute(conn, "SELECT id FROM nuvedra_rubric_criteria WHERE id=? AND rubric_id=?", (criterion_id, rubric_id)))
            if not found: raise HTTPException(404, "Rubric criterion not found.")
            count = rows(execute(conn, "SELECT COUNT(*) AS total FROM nuvedra_rubric_levels WHERE criterion_id=?", (criterion_id,)))
            execute(conn, "INSERT INTO nuvedra_rubric_levels (criterion_id,label,description,points,position,created_at) VALUES (?,?,?,?,?,?)", (criterion_id, label.strip(), description.strip(), points, int(count[0].get("total") or 0)+1, utcnow()))
        return RedirectResponse(f"{STUDIO_PREFIX}/rubrics/{rubric_id}", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/rubrics/{{rubric_id}}/attach", response_model=None)
    async def attach_rubric(rubric_id: int, request: Request, item_id: int = Form(...)):
        user = _user(request, f"{STUDIO_PREFIX}/rubrics/{rubric_id}")
        if isinstance(user, RedirectResponse): return user
        with db() as conn:
            rubric = _rubric(conn, rubric_id)
            course_id = int(rubric["course_id"])
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            item_course, _item, _module = academic_access.item_bundle(conn, item_id)
            if item_course != course_id: raise HTTPException(400, "Rubrics can only be attached to activities in the same course.")
            found = rows(execute(conn, "SELECT id FROM nuvedra_rubric_links WHERE item_id=?", (item_id,)))
            now = utcnow()
            if found:
                execute(conn, "UPDATE nuvedra_rubric_links SET rubric_id=?,created_by=?,created_at=? WHERE id=?", (rubric_id, user["email"], now, int(found[0]["id"])))
            else:
                execute(conn, "INSERT INTO nuvedra_rubric_links (rubric_id,item_id,created_by,created_at) VALUES (?,?,?,?)", (rubric_id, item_id, user["email"], now))
            audit(conn, user["email"], "rubric_attached", "item", str(item_id), f"rubric={rubric_id}", request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/rubrics/{rubric_id}", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/rubrics/{{rubric_id}}/duplicate", response_model=None)
    async def duplicate_rubric(rubric_id: int, request: Request, target_course_id: int = Form(...)):
        user = _user(request, f"{STUDIO_PREFIX}/rubrics/{rubric_id}")
        if isinstance(user, RedirectResponse): return user
        with db() as conn:
            source = _rubric(conn, rubric_id)
            academic_access.require_course_role(conn, int(source["course_id"]), user["email"], academic_access.AUTHOR_ROLES)
            academic_access.require_course_role(conn, target_course_id, user["email"], academic_access.AUTHOR_ROLES)
            now = utcnow()
            execute(conn, "INSERT INTO nuvedra_rubrics (course_id,title,description,status,created_by,created_at,updated_at) VALUES (?,?,?,'active',?,?,?)", (target_course_id, str(source.get("title") or "Rubric") + " (Copy)", source.get("description") or "", user["email"], now, now))
            new_id = int(rows(execute(conn, "SELECT id FROM nuvedra_rubrics WHERE course_id=? ORDER BY id DESC", (target_course_id,)))[0]["id"])
            for criterion in _criteria(conn, rubric_id):
                execute(conn, "INSERT INTO nuvedra_rubric_criteria (rubric_id,title,description,position,created_at) VALUES (?,?,?,?,?)", (new_id, criterion.get("title"), criterion.get("description"), int(criterion.get("position") or 1), now))
                new_criterion = int(rows(execute(conn, "SELECT id FROM nuvedra_rubric_criteria WHERE rubric_id=? ORDER BY id DESC", (new_id,)))[0]["id"])
                for level in criterion.get("levels", []):
                    execute(conn, "INSERT INTO nuvedra_rubric_levels (criterion_id,label,description,points,position,created_at) VALUES (?,?,?,?,?,?)", (new_criterion, level.get("label"), level.get("description"), float(level.get("points") or 0), int(level.get("position") or 1), now))
        return RedirectResponse(f"{STUDIO_PREFIX}/rubrics/{new_id}", status_code=303)

    @app.get(f"{STUDIO_PREFIX}/courses/{{course_id}}/outcomes", response_class=HTMLResponse, response_model=None)
    async def course_outcomes(course_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/outcomes")
        if isinstance(user, RedirectResponse): return user
        with db() as conn:
            course = academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            outcomes = rows(execute(conn, "SELECT * FROM nuvedra_outcomes WHERE course_id=? AND status='active' ORDER BY code,id", (course_id,)))

    @app.post(f"{STUDIO_PREFIX}/submissions/{{submission_id}}/rubric", response_model=None)
    async def save_rubric_grade(submission_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/submissions/{submission_id}/rubric")
        if isinstance(user, RedirectResponse): return user
        form = await request.form()
        with db() as conn:
            submission = _submission_bundle(conn, submission_id)
            course_id = int(submission["course_id"])
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            rubric = _linked_rubric(conn, int(submission["item_id"]))
            if not rubric: raise HTTPException(404, "No rubric is attached to this activity.")
            criteria = _criteria(conn, int(rubric["id"]))
            if not criteria: raise HTTPException(400, "The rubric has no criteria.")
            raw = 0.0; selections: list[tuple[int,int,float,str]] = []
            for criterion in criteria:
                criterion_id = int(criterion["id"])
                selected = str(form.get(f"level_{criterion_id}") or "").strip()
                if not selected.isdigit(): raise HTTPException(400, f"Select a performance level for {criterion.get('title') or 'each criterion'}.")
                level_id = int(selected)
                level = next((level for level in criterion.get("levels", []) if int(level["id"]) == level_id), None)
                if not level: raise HTTPException(400, "Selected rubric level is invalid.")
                points = float(level.get("points") or 0); raw += points
                selections.append((criterion_id, level_id, points, str(form.get(f"feedback_{criterion_id}") or "").strip()))
            possible = _rubric_max(criteria)
            if possible <= 0: raise HTTPException(400, "The rubric must contain positive possible points before grading.")
            item_points = float(submission.get("item_points") or 0)
            grade_points = (raw / possible) * item_points if item_points > 0 else raw
            grade_points = round(grade_points, 2)
            now = utcnow(); feedback = str(form.get("overall_feedback") or "").strip()
            evaluation_rows = rows(execute(conn, "SELECT id FROM nuvedra_rubric_evaluations WHERE submission_id=? AND rubric_id=?", (submission_id, int(rubric["id"]))))
            if evaluation_rows:
                evaluation_id = int(evaluation_rows[0]["id"])
                execute(conn, "UPDATE nuvedra_rubric_evaluations SET raw_score=?,possible_raw_score=?,grade_points=?,feedback=?,status='graded',graded_by=?,graded_at=?,updated_at=? WHERE id=?", (raw, possible, grade_points, feedback, user["email"], now, now, evaluation_id))
                execute(conn, "DELETE FROM nuvedra_rubric_scores WHERE evaluation_id=?", (evaluation_id,))
            else:
                execute(conn, "INSERT INTO nuvedra_rubric_evaluations (submission_id,rubric_id,raw_score,possible_raw_score,grade_points,feedback,status,graded_by,graded_at,updated_at) VALUES (?,?,?,?,?,?,'graded',?,?,?)", (submission_id, int(rubric["id"]), raw, possible, grade_points, feedback, user["email"], now, now))
                evaluation_id = int(rows(execute(conn, "SELECT id FROM nuvedra_rubric_evaluations WHERE submission_id=? AND rubric_id=?", (submission_id, int(rubric["id"]))))[0]["id"])
            for criterion_id, level_id, points, criterion_feedback in selections:
                execute(conn, "INSERT INTO nuvedra_rubric_scores (evaluation_id,criterion_id,level_id,points_awarded,feedback) VALUES (?,?,?,?,?)", (evaluation_id, criterion_id, level_id, points, criterion_feedback))
            grade = rows(execute(conn, "SELECT id FROM nuvedra_grades WHERE submission_id=?", (submission_id,)))
            if grade:
                execute(conn, "UPDATE nuvedra_grades SET points_awarded=?,feedback=?,status='graded',graded_by=?,graded_at=?,updated_at=? WHERE submission_id=?", (grade_points, feedback, user["email"], now, now, submission_id))
            else:
                execute(conn, "INSERT INTO nuvedra_grades (submission_id,points_awarded,feedback,status,graded_by,graded_at,updated_at) VALUES (?,?,?,'graded',?,?,?)", (submission_id, grade_points, feedback, user["email"], now, now))
            audit(conn, user["email"], "submission_rubric_graded", "submission", str(submission_id), f"rubric={int(rubric['id'])}; grade={grade_points:g}", request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/submissions/{submission_id}/rubric", status_code=303)
