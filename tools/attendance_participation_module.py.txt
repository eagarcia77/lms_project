STUDIO_PREFIX = "/faculty/studio"
ATTENDANCE_STATUSES = {"unmarked", "present", "absent", "late", "excused"}
PARTICIPATION_CATEGORIES = {"contribution", "discussion", "teamwork", "presentation", "other"}


def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def _insert_id(conn: Any, sql: str, params: tuple[Any, ...]) -> int:
    if database_url().startswith("postgres"):
        row = execute(conn, sql + " RETURNING id", params).fetchone()
        return int(row[0])
    cursor = execute(conn, sql, params)
    return int(cursor.lastrowid)


def ensure_attendance_participation_schema() -> None:
    pk = _pk()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_attendance_sessions (
            id {pk}, course_id INTEGER NOT NULL, title TEXT NOT NULL,
            session_date TEXT NOT NULL, notes TEXT, status TEXT NOT NULL DEFAULT 'active',
            created_by TEXT NOT NULL, created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_attendance_records (
            id {pk}, session_id INTEGER NOT NULL, student_email TEXT NOT NULL,
            status TEXT NOT NULL DEFAULT 'unmarked', minutes_late INTEGER NOT NULL DEFAULT 0,
            note TEXT, marked_by TEXT, updated_at TEXT NOT NULL,
            UNIQUE(session_id,student_email)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_participation_entries (
            id {pk}, course_id INTEGER NOT NULL, student_email TEXT NOT NULL,
            category TEXT NOT NULL DEFAULT 'contribution', points REAL NOT NULL DEFAULT 0,
            note TEXT, recorded_on TEXT NOT NULL, created_by TEXT NOT NULL,
            created_at TEXT NOT NULL
        )""")


def _course_students(conn: Any, course_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT lower(user_email) AS student_email
        FROM nexus_admin_enrollments
        WHERE course_id=? AND status='active' AND course_role='student'
        ORDER BY lower(user_email)""", (course_id,)))


def _session(conn: Any, session_id: int) -> dict[str, Any]:
    found = rows(execute(conn, """SELECT s.*,c.course_code,c.title AS course_title,c.status AS course_status
        FROM nuvedra_attendance_sessions s JOIN nexus_admin_courses c ON c.id=s.course_id
        WHERE s.id=?""", (session_id,)))
    if not found:
        raise HTTPException(404, "Attendance session not found.")
    return found[0]


def _author_session(conn: Any, session_id: int, email: str) -> dict[str, Any]:
    session = _session(conn, session_id)
    academic_access.require_course_role(conn, int(session["course_id"]), email, academic_access.AUTHOR_ROLES)
    return session


def _seed_session_records(conn: Any, session_id: int, course_id: int) -> None:
    now = utcnow()
    for student in _course_students(conn, course_id):
        email = str(student.get("student_email") or "").lower()
        if not email:
            continue
        existing = rows(execute(conn, "SELECT id FROM nuvedra_attendance_records WHERE session_id=? AND lower(student_email)=?", (session_id, email)))
        if not existing:
            execute(conn, """INSERT INTO nuvedra_attendance_records
                (session_id,student_email,status,minutes_late,note,marked_by,updated_at)
                VALUES (?,?,'unmarked',0,'',NULL,?)""", (session_id, email, now))


def attendance_summary_for_student(conn: Any, course_id: int, email: str) -> dict[str, Any]:
    counts = {status: 0 for status in ATTENDANCE_STATUSES}
    records = rows(execute(conn, """SELECT r.status
        FROM nuvedra_attendance_records r
        JOIN nuvedra_attendance_sessions s ON s.id=r.session_id
        WHERE s.course_id=? AND s.status='active' AND lower(r.student_email)=?""",
        (course_id, email.lower())))
    for record in records:
        status = str(record.get("status") or "unmarked")
        if status in counts:
            counts[status] += 1
    denominator = counts["present"] + counts["late"] + counts["absent"]
    attended = counts["present"] + counts["late"]
    rate = round((attended / denominator) * 100, 1) if denominator else None
    participation = rows(execute(conn, """SELECT COALESCE(SUM(points),0) AS points,COUNT(*) AS total
        FROM nuvedra_participation_entries WHERE course_id=? AND lower(student_email)=?""",
        (course_id, email.lower())))
    part_points = float(participation[0].get("points") or 0) if participation else 0.0
    part_entries = int(participation[0].get("total") or 0) if participation else 0
    return {
        **counts,
        "recorded_sessions": denominator + counts["excused"],
        "attendance_rate": rate,
        "participation_points": part_points,
        "participation_entries": part_entries,
    }


def _course_summaries(conn: Any, course_id: int) -> list[dict[str, Any]]:
    result = []
    for student in _course_students(conn, course_id):
        email = str(student.get("student_email") or "").lower()
        result.append({"email": email, **attendance_summary_for_student(conn, course_id, email)})
    return result


def _status_options(current: str) -> str:
    labels = [
        ("unmarked", "Unmarked"), ("present", "Present"), ("absent", "Absent"),
        ("late", "Late"), ("excused", "Excused"),
    ]
    return "".join(f'<option value="{value}"{" selected" if current == value else ""}>{label}</option>' for value, label in labels)

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/attendance/sessions", response_model=None)
    async def create_attendance_session(course_id: int, request: Request, title: str = Form(...), session_date: str = Form(...), notes: str = Form("")):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/attendance")
        if isinstance(user, RedirectResponse):
            return user
        clean_title = title.strip()
        clean_date = session_date.strip()
        if not clean_title or not clean_date:
            raise HTTPException(400, "Session title and date are required.")
        try:
            date.fromisoformat(clean_date)
        except ValueError as exc:
            raise HTTPException(400, "Attendance date must be a valid ISO date.") from exc
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            now = utcnow()
            session_id = _insert_id(conn, """INSERT INTO nuvedra_attendance_sessions
                (course_id,title,session_date,notes,status,created_by,created_at,updated_at)
                VALUES (?,?,?,?,'active',?,?,?)""", (course_id, clean_title, clean_date, notes.strip(), user["email"], now, now))
            _seed_session_records(conn, session_id, course_id)
            audit(conn, user["email"], "attendance_session_created", "attendance_session", str(session_id), clean_date, request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/attendance/sessions/{session_id}", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/attendance/sessions/{{session_id}}/records/{{record_id}}", response_model=None)
    async def save_attendance_record(session_id: int, record_id: int, request: Request, status: str = Form(...), minutes_late: int = Form(0), note: str = Form("")):
        user = _user(request, f"{STUDIO_PREFIX}/attendance/sessions/{session_id}")
        if isinstance(user, RedirectResponse):
            return user
        clean_status = status.strip().lower()
        if clean_status not in ATTENDANCE_STATUSES:
            raise HTTPException(400, "Invalid attendance status.")
        if minutes_late < 0 or minutes_late > 240:
            raise HTTPException(400, "Minutes late must be between 0 and 240.")
        with db() as conn:
            _author_session(conn, session_id, user["email"])
            record = rows(execute(conn, "SELECT * FROM nuvedra_attendance_records WHERE id=? AND session_id=?", (record_id, session_id)))
            if not record:
                raise HTTPException(404, "Attendance record not found.")
            execute(conn, """UPDATE nuvedra_attendance_records
                SET status=?,minutes_late=?,note=?,marked_by=?,updated_at=? WHERE id=?""",
                (clean_status, minutes_late if clean_status == "late" else 0, note.strip(), user["email"], utcnow(), record_id))
            audit(conn, user["email"], "attendance_record_updated", "attendance_record", str(record_id), f"status={clean_status}", request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/attendance/sessions/{session_id}", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/attendance/sessions/{{session_id}}/archive", response_model=None)
    async def archive_attendance_session(session_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/attendance/sessions/{session_id}")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            session = _author_session(conn, session_id, user["email"])
            execute(conn, "UPDATE nuvedra_attendance_sessions SET status='archived',updated_at=? WHERE id=?", (utcnow(), session_id))
            audit(conn, user["email"], "attendance_session_archived", "attendance_session", str(session_id), "", request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{int(session['course_id'])}/attendance", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/participation", response_model=None)
    async def record_participation(course_id: int, request: Request, student_email: str = Form(...), category: str = Form("contribution"), points: float = Form(1), note: str = Form("")):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/attendance")
        if isinstance(user, RedirectResponse):
            return user
        email = student_email.strip().lower()
        clean_category = category.strip().lower()
        if clean_category not in PARTICIPATION_CATEGORIES:
            raise HTTPException(400, "Invalid participation category.")
        if points < 0 or points > 10:
            raise HTTPException(400, "Participation points must be between 0 and 10.")
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            academic_access.require_course_role(conn, course_id, email, {"student"})
            execute(conn, """INSERT INTO nuvedra_participation_entries
                (course_id,student_email,category,points,note,recorded_on,created_by,created_at)
                VALUES (?,?,?,?,?,?,?,?)""", (course_id, email, clean_category, points, note.strip(), date.today().isoformat(), user["email"], utcnow()))
            audit(conn, user["email"], "participation_recorded", "course", str(course_id), f"student={email}; points={points}", request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{course_id}/attendance", status_code=303)

            return user
        with db() as conn:

        writer = csv.writer(stream)

        filename = f"{str(course.get('course_code') or 'course').replace(' ', '_')}-attendance-participation.csv"
        return Response(stream.getvalue(), media_type="text/csv; charset=utf-8", headers={"Content-Disposition": f'attachment; filename="{filename}"'})

    @app.get("/learn/courses/{course_id}/attendance", response_class=HTMLResponse, response_model=None)
    async def student_course_attendance(course_id: int, request: Request):
        user = _user(request, f"/learn/courses/{course_id}/attendance")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            course = academic_access.require_course_role(conn, course_id, user["email"], {"student"})
            summary = attendance_summary_for_student(conn, course_id, user["email"])
            records = rows(execute(conn, """SELECT s.session_date,s.title,r.status,r.minutes_late,r.note
                FROM nuvedra_attendance_records r JOIN nuvedra_attendance_sessions s ON s.id=r.session_id
                WHERE s.course_id=? AND s.status='active' AND lower(r.student_email)=?
                ORDER BY s.session_date DESC,s.id DESC""", (course_id, user["email"].lower())))
            participation = rows(execute(conn, """SELECT recorded_on,category,points,note
                FROM nuvedra_participation_entries WHERE course_id=? AND lower(student_email)=?
                ORDER BY recorded_on DESC,id DESC""", (course_id, user["email"].lower())))
        attendance_rows = "".join(f'''<tr><td>{academic_access.esc(row.get('session_date'))}</td><td>{academic_access.esc(row.get('title'))}</td><td>{academic_access.esc(row.get('status'))}</td><td>{int(row.get('minutes_late') or 0)}</td><td>{academic_access.esc(row.get('note'))}</td></tr>''' for row in records) or '<tr><td colspan="5">No attendance records yet.</td></tr>'
        participation_rows = "".join(f'''<tr><td>{academic_access.esc(row.get('recorded_on'))}</td><td>{academic_access.esc(row.get('category'))}</td><td>{float(row.get('points') or 0):.1f}</td><td>{academic_access.esc(row.get('note'))}</td></tr>''' for row in participation) or '<tr><td colspan="4">No participation entries yet.</td></tr>'
        rate = "—" if summary["attendance_rate"] is None else f"{summary['attendance_rate']:.1f}%"
        body = f'''{_assets()}<main class="studio-shell" data-studio-root data-testid="student-course-attendance-v1"><nav class="studio-breadcrumbs"><a href="/learn/attendance" data-i18n-en="My Attendance" data-i18n-es="Mi asistencia">My Attendance</a><span>/</span><a href="/learn/courses/{course_id}" data-i18n-en="Course" data-i18n-es="Curso">Course</a></nav><header class="studio-hero studio-hero--module"><div><p class="studio-eyebrow">{academic_access.esc(course.get('course_code'))}</p><h2>{academic_access.esc(course.get('title'))}</h2><p><strong>{rate}</strong> attendance rate · <strong>{summary['participation_points']:.1f}</strong> participation points</p></div></header><section class="studio-panel"><h3 data-i18n-en="Attendance history" data-i18n-es="Historial de asistencia">Attendance history</h3><div style="overflow:auto"><table><thead><tr><th>Date</th><th>Session</th><th>Status</th><th>Minutes late</th><th>Note</th></tr></thead><tbody>{attendance_rows}</tbody></table></div></section><section class="studio-panel"><h3 data-i18n-en="Participation evidence" data-i18n-es="Evidencia de participación">Participation evidence</h3><div style="overflow:auto"><table><thead><tr><th>Date</th><th>Category</th><th>Points</th><th>Note</th></tr></thead><tbody>{participation_rows}</tbody></table></div></section></main>'''
        return academic_access.portal_page("Course Attendance", body, user)
