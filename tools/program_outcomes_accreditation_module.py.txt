def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _insert_id(conn: Any, sql: str, params: tuple[Any, ...]) -> int:
    if database_url().startswith("postgres"):
        row = execute(conn, sql + " RETURNING id", params).fetchone()
        return int(row[0])
    cursor = execute(conn, sql, params)
    return int(cursor.lastrowid)


def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if user:
        return user
    admin = admin_console.session_user(request)
    if admin:
        return {"email": str(admin.get("email") or "").lower(), "name": admin.get("full_name") or admin.get("email")}
    return academic_access.login_redirect(return_to)


def _platform_role(request: Request) -> str:
    admin = admin_console.session_user(request)
    return str(admin.get("role") or "") if admin else ""


def ensure_program_outcomes_schema() -> None:
    pk = _pk()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_programs (
            id {pk}, program_code TEXT NOT NULL, title TEXT NOT NULL, description TEXT,
            status TEXT NOT NULL DEFAULT 'active', created_by TEXT NOT NULL,
            created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_program_members (
            id {pk}, program_id INTEGER NOT NULL, user_email TEXT NOT NULL,
            program_role TEXT NOT NULL DEFAULT 'reviewer', status TEXT NOT NULL DEFAULT 'active',
            created_by TEXT NOT NULL, created_at TEXT NOT NULL,
            UNIQUE(program_id,user_email)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_program_courses (
            id {pk}, program_id INTEGER NOT NULL, course_id INTEGER NOT NULL,
            status TEXT NOT NULL DEFAULT 'active', added_by TEXT NOT NULL, added_at TEXT NOT NULL,
            UNIQUE(program_id,course_id)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_program_outcomes (
            id {pk}, program_id INTEGER NOT NULL, code TEXT NOT NULL, title TEXT NOT NULL,
            description TEXT, status TEXT NOT NULL DEFAULT 'active', created_by TEXT NOT NULL,
            created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_program_outcome_links (
            id {pk}, program_outcome_id INTEGER NOT NULL, course_outcome_id INTEGER NOT NULL,
            curriculum_level TEXT NOT NULL DEFAULT 'introduced', weight REAL NOT NULL DEFAULT 1,
            created_by TEXT NOT NULL, created_at TEXT NOT NULL,
            UNIQUE(program_outcome_id,course_outcome_id)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_program_settings (
            id {pk}, program_id INTEGER NOT NULL UNIQUE, benchmark_threshold REAL NOT NULL DEFAULT 80,
            updated_by TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_program_snapshots (
            id {pk}, program_id INTEGER NOT NULL, label TEXT NOT NULL, snapshot_json TEXT NOT NULL,
            created_by TEXT NOT NULL, created_at TEXT NOT NULL
        )""")


def _has_author_course(conn: Any, email: str) -> bool:
    found = rows(execute(conn, """SELECT id FROM nexus_admin_enrollments
        WHERE lower(user_email)=? AND status='active'
          AND course_role IN ('instructor','teaching_assistant','course_builder','facilitator')
        LIMIT 1""", (email.lower(),)))
    return bool(found)


def _program(conn: Any, program_id: int) -> dict[str, Any]:
    found = rows(execute(conn, "SELECT * FROM nuvedra_programs WHERE id=?", (program_id,)))
    if not found:
        raise HTTPException(404, "Program not found.")
    return found[0]


def _program_access(conn: Any, request: Request, program_id: int, email: str) -> dict[str, Any]:
    platform_role = _platform_role(request)
    if platform_role in {"superadmin", "course_admin"}:
        return {"program_role": "coordinator", "platform_role": platform_role}
    if platform_role == "auditor":
        return {"program_role": "reviewer", "platform_role": platform_role}
    found = rows(execute(conn, """SELECT * FROM nuvedra_program_members
        WHERE program_id=? AND lower(user_email)=? AND status='active'""", (program_id, email.lower())))
    if not found:
        raise HTTPException(403, "You do not have access to this program assessment workspace.")
    return found[0]


def _require_manage(conn: Any, request: Request, program_id: int, email: str) -> dict[str, Any]:
    access = _program_access(conn, request, program_id, email)
    if str(access.get("program_role")) != "coordinator":
        raise HTTPException(403, "Program coordinator permission is required.")
    return access


def _program_courses(conn: Any, program_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT pc.id AS program_course_id,pc.course_id,pc.status AS mapping_status,
        c.course_code,c.title,c.term,c.status AS course_status
        FROM nuvedra_program_courses pc JOIN nexus_admin_courses c ON c.id=pc.course_id
        WHERE pc.program_id=? AND pc.status='active'
        ORDER BY c.course_code,c.term,c.id""", (program_id,)))


def _program_outcomes(conn: Any, program_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT * FROM nuvedra_program_outcomes
        WHERE program_id=? AND status='active' ORDER BY code,title,id""", (program_id,)))


def _settings(conn: Any, program_id: int) -> dict[str, Any]:
    found = rows(execute(conn, "SELECT * FROM nuvedra_program_settings WHERE program_id=?", (program_id,)))
    return found[0] if found else {"program_id": program_id, "benchmark_threshold": 80.0}


def _course_outcomes(conn: Any, program_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT o.id,o.course_id,o.code,o.title,o.description,c.course_code,c.title AS course_title,c.term
        FROM nuvedra_outcomes o
        JOIN nuvedra_program_courses pc ON pc.course_id=o.course_id AND pc.program_id=? AND pc.status='active'
        JOIN nexus_admin_courses c ON c.id=o.course_id
        WHERE o.status='active'
        ORDER BY c.course_code,o.code,o.title,o.id""", (program_id,)))


def _course_outcome_evidence(conn: Any, course_outcome_id: int) -> dict[str, Any]:
    outcome_rows = rows(execute(conn, """SELECT o.id,o.code,o.title,o.course_id,c.course_code,c.title AS course_title,c.term
        FROM nuvedra_outcomes o JOIN nexus_admin_courses c ON c.id=o.course_id WHERE o.id=?""", (course_outcome_id,)))
    if not outcome_rows:
        raise HTTPException(404, "Course outcome not found.")
    outcome = outcome_rows[0]
    evidence = rows(execute(conn, """SELECT i.id AS item_id,i.title AS item_title,i.points,
        g.points_awarded,g.status AS grade_status,s.status AS submission_status
        FROM nuvedra_outcome_links ol
        JOIN nexus_content_items i ON i.id=ol.item_id
        JOIN nuvedra_submissions s ON s.item_id=i.id AND s.status='submitted'
        JOIN nuvedra_grades g ON g.submission_id=s.id AND g.status='graded'
        WHERE ol.outcome_id=? AND i.points IS NOT NULL AND i.points>0 AND g.points_awarded IS NOT NULL""", (course_outcome_id,)))
    earned = 0.0
    possible = 0.0
    for row in evidence:
        maximum = float(row.get("points") or 0)
        earned += max(0.0, min(maximum, float(row.get("points_awarded") or 0)))
        possible += maximum
    outcome["attainment"] = None if possible <= 0 else (earned / possible) * 100.0
    outcome["graded_evidence"] = len(evidence)
    outcome["earned_points"] = earned
    outcome["possible_points"] = possible
    return outcome


def _program_outcome_mappings(conn: Any, program_outcome_id: int) -> list[dict[str, Any]]:
    data = rows(execute(conn, """SELECT pol.*,o.code AS course_outcome_code,o.title AS course_outcome_title,
        o.course_id,c.course_code,c.title AS course_title,c.term
        FROM nuvedra_program_outcome_links pol
        JOIN nuvedra_outcomes o ON o.id=pol.course_outcome_id
        JOIN nexus_admin_courses c ON c.id=o.course_id
        WHERE pol.program_outcome_id=?
        ORDER BY c.course_code,o.code,o.title""", (program_outcome_id,)))
    result: list[dict[str, Any]] = []
    for mapping in data:
        evidence = _course_outcome_evidence(conn, int(mapping["course_outcome_id"]))
        mapping["attainment"] = evidence["attainment"]
        mapping["graded_evidence"] = evidence["graded_evidence"]
        result.append(mapping)
    return result


def _program_outcome_summary(conn: Any, outcome: dict[str, Any], benchmark: float) -> dict[str, Any]:
    mappings = _program_outcome_mappings(conn, int(outcome["id"]))
    numerator = 0.0
    denominator = 0.0
    evidence_records = 0
    levels = {"introduced": 0, "reinforced": 0, "mastered": 0}
    for mapping in mappings:
        level = str(mapping.get("curriculum_level") or "introduced")
        if level in levels:
            levels[level] += 1
        evidence_records += int(mapping.get("graded_evidence") or 0)
        attainment = mapping.get("attainment")
        weight = float(mapping.get("weight") or 1)
        if attainment is not None and weight > 0:
            numerator += float(attainment) * weight
            denominator += weight
    attainment = None if denominator <= 0 else numerator / denominator
    if attainment is None:
        status = "No evidence"
    elif attainment + 1e-9 >= benchmark:
        status = "Meets benchmark"
    else:
        status = "Review gap"
    return {"outcome": outcome, "mappings": mappings, "attainment": attainment, "status": status,
            "graded_evidence": evidence_records, "levels": levels}


def _program_summary(conn: Any, program_id: int) -> dict[str, Any]:
    program = _program(conn, program_id)
    settings = _settings(conn, program_id)
    benchmark = float(settings.get("benchmark_threshold") or 80)
    courses = _program_courses(conn, program_id)
    outcomes = [_program_outcome_summary(conn, outcome, benchmark) for outcome in _program_outcomes(conn, program_id)]
    return {"program": program, "settings": settings, "courses": courses, "outcomes": outcomes}


def _matrix_cells(summary: dict[str, Any]) -> dict[tuple[int, int], str]:
    cells: dict[tuple[int, int], set[str]] = {}
    for entry in summary["outcomes"]:
        po_id = int(entry["outcome"]["id"])
        for mapping in entry["mappings"]:
            key = (po_id, int(mapping["course_id"]))
            cells.setdefault(key, set()).add(LEVEL_LABEL.get(str(mapping.get("curriculum_level")), "?"))
    return {key: " / ".join(level for level in ("I", "R", "M") if level in values) for key, values in cells.items()}


def _snapshot_payload(conn: Any, program_id: int) -> dict[str, Any]:
    summary = _program_summary(conn, program_id)
    return {
        "version": 1,
        "captured_at": utcnow(),
        "program": {key: summary["program"].get(key) for key in ("program_code", "title", "description", "status")},
        "benchmark_threshold": float(summary["settings"].get("benchmark_threshold") or 80),
        "courses": [{key: course.get(key) for key in ("course_code", "title", "term", "course_status")} for course in summary["courses"]],
        "outcomes": [
            {
                "code": entry["outcome"].get("code"), "title": entry["outcome"].get("title"),
                "attainment": entry["attainment"], "status": entry["status"], "graded_evidence": entry["graded_evidence"],
                "coverage": entry["levels"],
                "mappings": [
                    {"course_code": mapping.get("course_code"), "course_outcome_code": mapping.get("course_outcome_code"),
                     "course_outcome_title": mapping.get("course_outcome_title"), "curriculum_level": mapping.get("curriculum_level"),
                     "weight": mapping.get("weight"), "attainment": mapping.get("attainment"), "graded_evidence": mapping.get("graded_evidence")}
                    for mapping in entry["mappings"]
                ],
            }
            for entry in summary["outcomes"]
        ],
    }

    @app.post("/faculty/programs", response_model=None)
    async def create_program(request: Request, program_code: str = Form(...), title: str = Form(...), description: str = Form("")):
        user = _user(request, "/faculty/programs")
        if isinstance(user, RedirectResponse): return user
        code = program_code.strip().upper(); clean_title = title.strip()
        if not code or not clean_title: raise HTTPException(400, "Program code and title are required.")
        with db() as conn:
            if _platform_role(request) not in {"superadmin", "course_admin"} and not _has_author_course(conn, user["email"]):
                raise HTTPException(403, "Create a program workspace only when you have course-author responsibilities or platform administration permission.")
            if rows(execute(conn, "SELECT id FROM nuvedra_programs WHERE upper(program_code)=upper(?) AND status='active'", (code,))):
                raise HTTPException(409, "An active program workspace already uses this code.")
            now = utcnow()
            program_id = _insert_id(conn, "INSERT INTO nuvedra_programs (program_code,title,description,status,created_by,created_at,updated_at) VALUES (?,?,?,'active',?,?,?)", (code, clean_title, description.strip(), user["email"], now, now))
            execute(conn, "INSERT INTO nuvedra_program_members (program_id,user_email,program_role,status,created_by,created_at) VALUES (?,?, 'coordinator','active',?,?)", (program_id, user["email"].lower(), user["email"], now))
            audit(conn, user["email"], "program_assessment_workspace_created", "program", str(program_id), code, request.client.host if request.client else "")
        return RedirectResponse(f"/faculty/programs/{program_id}", status_code=303)

    @app.post("/faculty/programs/{program_id}/courses", response_model=None)
    async def add_program_course(program_id: int, request: Request, course_id: int = Form(...)):
        user = _user(request, f"/faculty/programs/{program_id}")
        if isinstance(user, RedirectResponse): return user
        with db() as conn:

                execute(conn, "INSERT INTO nuvedra_program_courses (program_id,course_id,status,added_by,added_at) VALUES (?,?,'active',?,?)", (program_id, course_id, user["email"], utcnow()))

    @app.post("/faculty/programs/{program_id}/alignments", response_model=None)
    async def create_program_alignment(program_id: int, request: Request, program_outcome_id: int = Form(...), course_outcome_id: int = Form(...), curriculum_level: str = Form("introduced"), weight: float = Form(1)):
        user = _user(request, f"/faculty/programs/{program_id}")
        if isinstance(user, RedirectResponse): return user
        if curriculum_level not in CURRICULUM_LEVELS: raise HTTPException(400, "Unsupported curriculum level.")
        if weight < 0.1 or weight > 10: raise HTTPException(400, "Evidence weight must be between 0.1 and 10.")
        with db() as conn:
            _require_manage(conn, request, program_id, user["email"])
            po = rows(execute(conn, "SELECT id FROM nuvedra_program_outcomes WHERE id=? AND program_id=? AND status='active'", (program_outcome_id, program_id)))
            co = rows(execute(conn, """SELECT o.id,o.course_id FROM nuvedra_outcomes o JOIN nuvedra_program_courses pc ON pc.course_id=o.course_id
                WHERE o.id=? AND pc.program_id=? AND pc.status='active' AND o.status='active'""", (course_outcome_id, program_id)))
            if not po or not co: raise HTTPException(400, "Both outcomes must belong to this active program curriculum.")
            existing = rows(execute(conn, "SELECT id FROM nuvedra_program_outcome_links WHERE program_outcome_id=? AND course_outcome_id=?", (program_outcome_id, course_outcome_id)))
            if existing:
                execute(conn, "UPDATE nuvedra_program_outcome_links SET curriculum_level=?,weight=?,created_by=?,created_at=? WHERE id=?", (curriculum_level, weight, user["email"], utcnow(), int(existing[0]["id"])))
            else:
                execute(conn, "INSERT INTO nuvedra_program_outcome_links (program_outcome_id,course_outcome_id,curriculum_level,weight,created_by,created_at) VALUES (?,?,?,?,?,?)", (program_outcome_id, course_outcome_id, curriculum_level, weight, user["email"], utcnow()))
            audit(conn, user["email"], "program_outcome_alignment_saved", "program_outcome", str(program_outcome_id), f"course_outcome={course_outcome_id};level={curriculum_level};weight={weight:g}", request.client.host if request.client else "")
        return RedirectResponse(f"/faculty/programs/{program_id}", status_code=303)
