def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _b64u(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")


def _b64ud(value: str) -> bytes:
    return base64.urlsafe_b64decode(value + "=" * (-len(value) % 4))


def _json_b64(value: Any) -> str:
    return _b64u(json.dumps(value, separators=(",", ":"), sort_keys=True).encode("utf-8"))


def _insert_id(conn: Any, sql: str, params: tuple[Any, ...]) -> int:
    if database_url().startswith("postgres"):
        row = execute(conn, sql + " RETURNING id", params).fetchone()
        return int(row[0])
    cursor = execute(conn, sql, params)
    return int(cursor.lastrowid)


def _safe_https_url(value: str, label: str) -> str:
    clean = value.strip()
    parsed = urllib.parse.urlparse(clean)
    if parsed.scheme != "https" or not parsed.hostname or parsed.username or parsed.password:
        raise HTTPException(400, f"{label} must be an HTTPS URL without embedded credentials.")
    host = parsed.hostname.lower()
    if host in {"localhost", "localhost.localdomain"} or host.endswith(".local"):
        raise HTTPException(400, f"{label} cannot target a local host.")
    try:
        ip = ipaddress.ip_address(host.strip("[]"))
    except ValueError:
        ip = None
    if ip is not None and (ip.is_private or ip.is_loopback or ip.is_link_local or ip.is_reserved or ip.is_multicast):
        raise HTTPException(400, f"{label} cannot target a private or local address.")
    return clean


def _redirect_uris(value: str) -> list[str]:
    raw = value.replace(",", "\n").splitlines()
    uris = []
    for entry in raw:
        if entry.strip():
            uris.append(_safe_https_url(entry, "Redirect URI"))
    if not uris:
        raise HTTPException(400, "At least one HTTPS redirect URI is required.")
    return list(dict.fromkeys(uris))

def _platform_issuer(request: Request) -> str:
    configured = os.getenv("NUVEDRA_LTI13_ISSUER", "").strip().rstrip("/")
    return configured or str(request.base_url).rstrip("/")


def _absolute(request: Request, path: str) -> str:
    return str(request.base_url).rstrip("/") + path


def _subject(client_id: str, email: str) -> str:
    return hashlib.sha256(f"{client_id}|{email.lower()}".encode("utf-8")).hexdigest()[:40]


def _parse_iso(value: Any) -> datetime | None:
    clean = str(value or "").strip()
    if not clean:
        return None
    try:
        parsed = datetime.fromisoformat(clean.replace("Z", "+00:00"))
    except ValueError:
        return None
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=timezone.utc)
    return parsed.astimezone(timezone.utc)

def _ensure_platform_key(conn: Any) -> dict[str, Any]:
    found = rows(execute(conn, "SELECT * FROM nuvedra_lti13_platform_keys WHERE status='active' ORDER BY id DESC LIMIT 1"))
    if found:
        return found[0]
    private = rsa.generate_private_key(public_exponent=65537, key_size=2048)
    private_pem = private.private_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PrivateFormat.PKCS8,
        encryption_algorithm=serialization.NoEncryption(),
    ).decode("ascii")
    numbers = private.public_key().public_numbers()
    kid = secrets.token_urlsafe(18)
    jwk = {
        "kty": "RSA", "use": "sig", "alg": "RS256", "kid": kid,
        "n": _b64u(numbers.n.to_bytes((numbers.n.bit_length() + 7) // 8, "big")),
        "e": _b64u(numbers.e.to_bytes((numbers.e.bit_length() + 7) // 8, "big")),
    }
    now = utcnow()
    _insert_id(conn, "INSERT INTO nuvedra_lti13_platform_keys (kid,private_pem,public_jwk_json,status,created_at) VALUES (?,?,?,'active',?)", (kid, private_pem, json.dumps(jwk, separators=(",", ":")), now))
    return rows(execute(conn, "SELECT * FROM nuvedra_lti13_platform_keys WHERE kid=?", (kid,)))[0]


def _sign_jwt(conn: Any, payload: dict[str, Any]) -> str:
    key = _ensure_platform_key(conn)
    header = {"alg": "RS256", "typ": "JWT", "kid": key["kid"]}
    signing_input = f"{_json_b64(header)}.{_json_b64(payload)}"
    private = serialization.load_pem_private_key(str(key["private_pem"]).encode("ascii"), password=None)
    signature = private.sign(signing_input.encode("ascii"), padding.PKCS1v15(), hashes.SHA256())
    return signing_input + "." + _b64u(signature)


def _jwt_parts(token: str) -> tuple[dict[str, Any], dict[str, Any], bytes, bytes]:
    parts = token.split(".")
    if len(parts) != 3:
        raise HTTPException(401, "Invalid JWT.")
    try:
        header = json.loads(_b64ud(parts[0]))
        payload = json.loads(_b64ud(parts[1]))
        signature = _b64ud(parts[2])
    except Exception as exc:
        raise HTTPException(401, "Invalid JWT encoding.") from exc
    return header, payload, f"{parts[0]}.{parts[1]}".encode("ascii"), signature


def _fetch_jwks(url: str) -> dict[str, Any]:
    safe = _safe_https_url(url, "JWKS URL")
    try:
        response = httpx.get(safe, timeout=5.0, follow_redirects=False, headers={"Accept": "application/json"})
        response.raise_for_status()
        data = response.json()
    except Exception as exc:
        raise HTTPException(502, "Unable to retrieve the external tool JWKS.") from exc
    if not isinstance(data, dict) or not isinstance(data.get("keys"), list):
        raise HTTPException(502, "External tool JWKS is invalid.")
    return data


def _rsa_public_key(jwk: dict[str, Any]):
    if jwk.get("kty") != "RSA" or not jwk.get("n") or not jwk.get("e"):
        raise HTTPException(401, "Unsupported external tool signing key.")
    n = int.from_bytes(_b64ud(str(jwk["n"])), "big")
    e = int.from_bytes(_b64ud(str(jwk["e"])), "big")
    return rsa.RSAPublicNumbers(e, n).public_key()


def _audience_matches(value: Any, expected: str) -> bool:
    if isinstance(value, str):
        return value == expected
    if isinstance(value, list):
        return expected in value
    return False


def _verify_tool_jwt(token: str, tool: dict[str, Any], *, audience: str, issuer: str | None = None) -> dict[str, Any]:
    header, payload, signed, signature = _jwt_parts(token)
    if header.get("alg") != "RS256":
        raise HTTPException(401, "Only RS256 tool assertions are supported.")
    keys = _fetch_jwks(str(tool["jwks_url"])).get("keys", [])
    kid = str(header.get("kid") or "")
    candidates = [key for key in keys if not kid or str(key.get("kid") or "") == kid]
    if not candidates:
        raise HTTPException(401, "External tool signing key was not found.")
    verified = False
    for jwk in candidates:
        try:
            _rsa_public_key(jwk).verify(signature, signed, padding.PKCS1v15(), hashes.SHA256())
            verified = True
            break
        except Exception:
            continue
    if not verified:
        raise HTTPException(401, "External tool JWT signature is invalid.")
    now = int(time.time())
    try:
        exp = int(payload.get("exp"))
    except Exception as exc:
        raise HTTPException(401, "External tool JWT is missing a valid expiration.") from exc
    if exp < now - 30 or exp > now + 600:
        raise HTTPException(401, "External tool JWT expiration is invalid.")
    if payload.get("nbf") is not None and int(payload["nbf"]) > now + 30:
        raise HTTPException(401, "External tool JWT is not yet valid.")
    if issuer is not None and str(payload.get("iss") or "") != issuer:
        raise HTTPException(401, "External tool JWT issuer is invalid.")
    if not _audience_matches(payload.get("aud"), audience):
        raise HTTPException(401, "External tool JWT audience is invalid.")
    return payload

def _role_claim(role: str) -> list[str]:
    base = "http://purl.imsglobal.org/vocab/lis/v2/membership#"
    if role == "student": return [base + "Learner"]
    if role == "observer": return [base + "Observer"]
    return [base + "Instructor"]


def _custom_parameters(value: str) -> dict[str, str]:
    result: dict[str, str] = {}
    for line in str(value or "").splitlines():
        if "=" not in line:
            continue
        key, val = line.split("=", 1)
        key = key.strip()
        if key and len(key) <= 120:
            result[key] = val.strip()[:1000]
    return result

def _launch_redirect(conn: Any, request: Request, tool: dict[str, Any], user: dict[str, Any], *, message_type: str, target_link_uri: str, module_id: int | None = None, item_id: int | None = None) -> RedirectResponse:
    token = secrets.token_urlsafe(32)
    now_dt = datetime.now(timezone.utc)
    created = now_dt.isoformat()
    expires = datetime.fromtimestamp(now_dt.timestamp() + 600, tz=timezone.utc).isoformat()
    _insert_id(conn, """INSERT INTO nuvedra_lti13_launches
        (launch_token,tool_id,course_id,module_id,item_id,user_email,message_type,target_link_uri,created_at,expires_at)
        VALUES (?,?,?,?,?,?,?,?,?,?)""", (token, int(tool["id"]), int(tool["course_id"]), module_id, item_id, user["email"].lower(), message_type, target_link_uri, created, expires))
    params = {
        "iss": _platform_issuer(request),
        "login_hint": token,
        "target_link_uri": target_link_uri,
        "lti_message_hint": token,
        "client_id": str(tool["client_id"]),
        "lti_deployment_id": str(tool["deployment_id"]),
    }
    location = str(tool["auth_login_url"]) + ("&" if "?" in str(tool["auth_login_url"]) else "?") + urllib.parse.urlencode(params)
    return RedirectResponse(location, status_code=303, headers={"Cache-Control": "no-store"})

def _token_record(conn: Any, request: Request, required: set[str]) -> dict[str, Any]:
    auth = request.headers.get("authorization", "")
    if not auth.lower().startswith("bearer "):
        raise HTTPException(401, "Bearer token required.", headers={"WWW-Authenticate": "Bearer"})
    raw = auth.split(" ", 1)[1].strip()
    digest = hashlib.sha256(raw.encode()).hexdigest()
    found = rows(execute(conn, "SELECT * FROM nuvedra_lti13_tokens WHERE token_hash=?", (digest,)))
    if not found:
        raise HTTPException(401, "Invalid bearer token.", headers={"WWW-Authenticate": "Bearer"})
    record = found[0]
    expires = _parse_iso(record.get("expires_at"))
    if expires is None or expires <= datetime.now(timezone.utc):
        raise HTTPException(401, "Bearer token expired.", headers={"WWW-Authenticate": "Bearer"})
    scopes = set(str(record.get("scopes") or "").split())
    if not (required & scopes):
        raise HTTPException(403, "The access token does not include the required scope.")
    return record

def _student_email_for_subject(conn: Any, course_id: int, client_id: str, user_id: str) -> str | None:
    enrolled = rows(execute(conn, """SELECT lower(user_email) AS email FROM nexus_admin_enrollments
        WHERE course_id=? AND status='active' AND course_role='student'""", (course_id,)))
    for row in enrolled:
        email = str(row["email"])
        if _subject(client_id, email) == user_id:
            return email
    return None


def _sync_ags_grade(conn: Any, lineitem: dict[str, Any], user_id: str, score_given: float, score_maximum: float, actor: str) -> None:
    item_id = lineitem.get("item_id")
    if item_id in (None, "") or score_maximum <= 0:
        return
    email = _student_email_for_subject(conn, int(lineitem["course_id"]), str(lineitem["client_id"]), user_id)
    if not email:
        return
    item = rows(execute(conn, "SELECT points FROM nexus_content_items WHERE id=?", (int(item_id),)))
    if not item:
        return
    possible = float(item[0].get("points") or 0)
    if possible <= 0:
        possible = float(score_maximum)
        execute(conn, "UPDATE nexus_content_items SET points=?,updated_at=? WHERE id=?", (possible, utcnow(), int(item_id)))
    awarded = max(0.0, min(possible, (float(score_given) / float(score_maximum)) * possible))
    now = utcnow()
    submission = rows(execute(conn, "SELECT id FROM nuvedra_submissions WHERE item_id=? AND lower(student_email)=?", (int(item_id), email)))
    if submission:
        submission_id = int(submission[0]["id"])
        execute(conn, "UPDATE nuvedra_submissions SET status='submitted',updated_at=? WHERE id=?", (now, submission_id))
    else:
        submission_id = _insert_id(conn, """INSERT INTO nuvedra_submissions
            (item_id,student_email,response_text,response_url,status,submitted_at,updated_at)
            VALUES (?,?,?,'','submitted',?,?)""", (int(item_id), email, "LTI 1.3 / AGS score received.", now, now))
    grade = rows(execute(conn, "SELECT id FROM nuvedra_grades WHERE submission_id=?", (submission_id,)))
    if grade:
        execute(conn, "UPDATE nuvedra_grades SET points_awarded=?,feedback=?,status='graded',graded_by=?,graded_at=?,updated_at=? WHERE submission_id=?", (awarded, "Grade synchronized through LTI 1.3 Assignment and Grade Services.", actor, now, now, submission_id))
    else:
        execute(conn, "INSERT INTO nuvedra_grades (submission_id,points_awarded,feedback,status,graded_by,graded_at,updated_at) VALUES (?,?,?,?,?,?,?)", (submission_id, awarded, "Grade synchronized through LTI 1.3 Assignment and Grade Services.", "graded", actor, now, now))
    _upsert_progress(conn, int(item_id), email, True)

                (tool_id,course_id,module_id,item_id,title,target_link_uri,custom_parameters,status,created_at,updated_at)

    @app.api_route("/lti13/authorize", methods=["GET", "POST"], response_class=HTMLResponse, response_model=None)
    async def lti13_authorize(request: Request):
        user = _user(request, "/portal")
        if isinstance(user, RedirectResponse): return user
        params = dict(request.query_params)
        if request.method == "POST": params.update({str(k): str(v) for k, v in (await request.form()).items()})
        login_hint = str(params.get("login_hint") or "")
        message_hint = str(params.get("lti_message_hint") or "")
        client_id = str(params.get("client_id") or "")
        redirect_uri = str(params.get("redirect_uri") or "")
        state = str(params.get("state") or "")
        nonce = str(params.get("nonce") or "")
        if not login_hint or login_hint != message_hint or not client_id or not redirect_uri or not state or not nonce:
            raise HTTPException(400, "OIDC authorization request is incomplete.")
        if "id_token" not in str(params.get("response_type") or "") or "openid" not in str(params.get("scope") or ""):
            raise HTTPException(400, "OIDC authorization request must request an OpenID id_token.")
        with db() as conn:
            launches = rows(execute(conn, "SELECT * FROM nuvedra_lti13_launches WHERE launch_token=?", (login_hint,)))
            if not launches: raise HTTPException(400, "OIDC login hint is invalid.")
            launch = launches[0]
            expiry = _parse_iso(launch.get("expires_at"))
            if expiry is None or expiry <= datetime.now(timezone.utc): raise HTTPException(400, "OIDC login hint expired.")
            if launch.get("authorized_at"): raise HTTPException(409, "OIDC login hint has already been used.")
            tool = _tool(conn, int(launch["tool_id"]))
            if client_id != str(tool["client_id"]): raise HTTPException(400, "OIDC Client ID does not match the launch.")
            if redirect_uri not in _registered_redirects(tool): raise HTTPException(400, "OIDC redirect URI is not registered.")
            if user["email"].lower() != str(launch["user_email"]).lower(): raise HTTPException(403, "OIDC launch belongs to a different signed-in user.")
            access = academic_access.require_course_role(conn, int(tool["course_id"]), user["email"], academic_access.AUTHOR_ROLES | academic_access.STUDENT_ROLES)
            now = int(time.time())
            payload: dict[str, Any] = {
                "iss": _platform_issuer(request), "aud": str(tool["client_id"]), "sub": _subject(str(tool["client_id"]), user["email"]),
                "iat": now, "exp": now + 300, "nonce": nonce,
                LTI_CLAIM + "deployment_id": str(tool["deployment_id"]),
                LTI_CLAIM + "message_type": str(launch["message_type"]),
                LTI_CLAIM + "version": LTI_VERSION,
                LTI_CLAIM + "target_link_uri": str(launch["target_link_uri"]),
                LTI_CLAIM + "roles": _role_claim(str(access.get("course_role") or "")),
                LTI_CLAIM + "context": {"id": f"nuvedra-course-{int(tool['course_id'])}", "label": str(tool.get("course_code") or "course"), "title": str(tool.get("course_title") or "Course")},
                LTI_CLAIM + "launch_presentation": {"return_url": _absolute(request, f"/learn/courses/{int(tool['course_id'])}")},
            }
            if int(tool.get("share_email") or 0): payload["email"] = user["email"].lower()
            if str(launch["message_type"]) == "LtiResourceLinkRequest":
                resource_rows = rows(execute(conn, "SELECT * FROM nuvedra_lti13_resources WHERE tool_id=? AND item_id=?", (int(tool["id"]), int(launch["item_id"]))))
                if not resource_rows: raise HTTPException(404, "LTI resource link no longer exists.")
                resource = resource_rows[0]
                payload[LTI_CLAIM + "resource_link"] = {"id": f"nuvedra-item-{int(resource['item_id'])}", "title": str(resource["title"])}
                custom = _custom_parameters(str(resource.get("custom_parameters") or ""))
                if custom: payload[LTI_CLAIM + "custom"] = custom
                lineitems = rows(execute(conn, "SELECT * FROM nuvedra_lti13_lineitems WHERE tool_id=? AND item_id=? ORDER BY id LIMIT 1", (int(tool["id"]), int(resource["item_id"]))))
                endpoint: dict[str, Any] = {"scope": [AGS_SCOPE_LINEITEM, AGS_SCOPE_LINEITEM_READ, AGS_SCOPE_RESULT, AGS_SCOPE_SCORE], "lineitems": _absolute(request, f"/lti13/ags/tools/{int(tool['id'])}/lineitems")}
                if lineitems: endpoint["lineitem"] = _absolute(request, f"/lti13/ags/lineitems/{int(lineitems[0]['id'])}")
                payload[AGS_CLAIM + "endpoint"] = endpoint
            elif str(launch["message_type"]) == "LtiDeepLinkingRequest":
                if str(access.get("course_role")) not in academic_access.AUTHOR_ROLES: raise HTTPException(403, "Deep Linking requires instructor access.")
                payload[DL_CLAIM + "deep_linking_settings"] = {
                    "deep_link_return_url": _absolute(request, f"/lti13/deep-link/return/{int(tool['id'])}"),
                    "accept_types": ["ltiResourceLink", "link"],
                    "accept_presentation_document_targets": ["iframe", "window"],
                    "accept_multiple": True, "auto_create": True, "title": "NUVEDRA course content", "data": login_hint,
                }
            else:
                raise HTTPException(400, "Unsupported LTI message type.")
            token = _sign_jwt(conn, payload)
            execute(conn, "UPDATE nuvedra_lti13_launches SET authorized_at=? WHERE id=?", (utcnow(), int(launch["id"])))
        form = f'''<!doctype html><html><head><meta charset="utf-8"><meta name="referrer" content="no-referrer"><title>LTI 1.3 launch</title></head><body><main><h1>Launching learning tool</h1><form id="lti13" method="post" action="{academic_access.esc(redirect_uri, attr=True)}"><input type="hidden" name="id_token" value="{academic_access.esc(token, attr=True)}"><input type="hidden" name="state" value="{academic_access.esc(state, attr=True)}"><noscript><button type="submit">Continue</button></noscript></form></main><script>document.getElementById('lti13').submit();</script></body></html>'''
        return HTMLResponse(form, headers={"Cache-Control": "no-store", "Referrer-Policy": "no-referrer"})

    @app.post("/lti13/token", response_model=None)
    async def lti13_token(request: Request, grant_type: str = Form(...), client_assertion_type: str = Form(...), client_assertion: str = Form(...), scope: str = Form(...)):
        if grant_type != "client_credentials" or client_assertion_type != "urn:ietf:params:oauth:client-assertion-type:jwt-bearer":
            raise HTTPException(400, "Unsupported OAuth grant.")
        _, unverified, _, _ = _jwt_parts(client_assertion)
        client_id = str(unverified.get("iss") or "")
        if not client_id or str(unverified.get("sub") or "") != client_id: raise HTTPException(401, "Client assertion issuer and subject must match the Client ID.")
        with db() as conn:
            tool = _tool_by_client(conn, client_id)
            payload = _verify_tool_jwt(client_assertion, tool, audience=_absolute(request, "/lti13/token"), issuer=client_id)
            jti = str(payload.get("jti") or "")
            if not jti: raise HTTPException(401, "Client assertion must include jti.")
            jti_hash = hashlib.sha256(jti.encode()).hexdigest()
            if rows(execute(conn, "SELECT id FROM nuvedra_lti13_assertion_jtis WHERE jti_hash=?", (jti_hash,))): raise HTTPException(401, "Client assertion has already been used.")
            requested = set(scope.split()) & ALLOWED_SCOPES
            if not requested: raise HTTPException(400, "No supported LTI Advantage scope was requested.")
            now = datetime.now(timezone.utc); expires = datetime.fromtimestamp(now.timestamp() + 3600, tz=timezone.utc)
            execute(conn, "INSERT INTO nuvedra_lti13_assertion_jtis (jti_hash,client_id,expires_at) VALUES (?,?,?)", (jti_hash, client_id, datetime.fromtimestamp(int(payload["exp"]), tz=timezone.utc).isoformat()))
            raw = secrets.token_urlsafe(40); digest = hashlib.sha256(raw.encode()).hexdigest()
            _insert_id(conn, "INSERT INTO nuvedra_lti13_tokens (token_hash,client_id,scopes,created_at,expires_at) VALUES (?,?,?,?,?)", (digest, client_id, " ".join(sorted(requested)), now.isoformat(), expires.isoformat()))
        return JSONResponse({"access_token": raw, "token_type": "Bearer", "expires_in": 3600, "scope": " ".join(sorted(requested))}, headers={"Cache-Control": "no-store"})

        if not user_id: raise HTTPException(400, "userId is required.")
        score_given = body.get("scoreGiven"); score_max = body.get("scoreMaximum")
        try:
            given = float(score_given) if score_given is not None else None
            maximum = float(score_max) if score_max is not None else None
        except (TypeError, ValueError) as exc:
            raise HTTPException(400, "scoreGiven and scoreMaximum must be numeric when supplied.") from exc
        with db() as conn:
            token = _token_record(conn, request, {AGS_SCOPE_SCORE})
            lineitem = _lineitem(conn, lineitem_id)
            if token.get("client_id") != lineitem.get("client_id"): raise HTTPException(403, "Token does not belong to this line item.")
            now = utcnow(); activity = str(body.get("activityProgress") or ""); grading = str(body.get("gradingProgress") or ""); comment = str(body.get("comment") or "")[:4000]; timestamp = str(body.get("timestamp") or now)[:120]
            found = rows(execute(conn, "SELECT id FROM nuvedra_lti13_scores WHERE lineitem_id=? AND user_id=?", (lineitem_id, user_id)))
            if found:
                execute(conn, "UPDATE nuvedra_lti13_scores SET score_given=?,score_maximum=?,activity_progress=?,grading_progress=?,comment=?,timestamp=?,updated_at=? WHERE id=?", (given, maximum, activity, grading, comment, timestamp, now, int(found[0]["id"])))
            else:
                _insert_id(conn, "INSERT INTO nuvedra_lti13_scores (lineitem_id,user_id,score_given,score_maximum,activity_progress,grading_progress,comment,timestamp,updated_at) VALUES (?,?,?,?,?,?,?,?,?)", (lineitem_id, user_id, given, maximum, activity, grading, comment, timestamp, now))
            if given is not None and maximum is not None and maximum > 0 and grading == "FullyGraded":
                _sync_ags_grade(conn, lineitem, user_id, given, maximum, f"lti13:{lineitem['client_id']}")
            elif lineitem.get("item_id") not in (None, "") and (activity == "Completed" or grading == "FullyGraded"):
                email = _student_email_for_subject(conn, int(lineitem["course_id"]), str(lineitem["client_id"]), user_id)
                if email: _upsert_progress(conn, int(lineitem["item_id"]), email, True)
            audit(conn, f"lti13:{lineitem['client_id']}", "lti13_ags_score_received", "lineitem", str(lineitem_id), grading, request.client.host if request.client else "")
        return JSONResponse({}, status_code=200)
