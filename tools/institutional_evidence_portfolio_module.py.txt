MAX_UPLOAD_BYTES = 25 * 1024 * 1024
EVIDENCE_TYPES = {"report", "policy", "rubric", "minutes", "survey", "dataset", "curriculum", "faculty", "student_work_sample", "assessment", "improvement", "other"}
ALLOWED_EXTENSIONS = {".pdf", ".doc", ".docx", ".ppt", ".pptx", ".xls", ".xlsx", ".csv", ".txt", ".png", ".jpg", ".jpeg", ".webp", ".zip"}


def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _blob_type() -> str:
    return "BYTEA" if database_url().startswith("postgres") else "BLOB"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _insert_id(conn: Any, sql: str, params: tuple[Any, ...]) -> int:
    if database_url().startswith("postgres"):
        row = execute(conn, sql + " RETURNING id", params).fetchone()
        return int(row[0])
    cursor = execute(conn, sql, params)
    return int(cursor.lastrowid)


def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    return program_outcomes._user(request, return_to)


def ensure_evidence_portfolio_schema() -> None:
    pk = _pk()
    blob = _blob_type()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_evidence_assets (
            id {pk}, title TEXT NOT NULL, description TEXT, evidence_type TEXT NOT NULL,
            tags TEXT, status TEXT NOT NULL DEFAULT 'active', created_by TEXT NOT NULL,
            created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_evidence_versions (
            id {pk}, asset_id INTEGER NOT NULL, version_no INTEGER NOT NULL,
            file_name TEXT, mime_type TEXT, file_size INTEGER NOT NULL DEFAULT 0,
            file_bytes {blob}, source_url TEXT, change_note TEXT,
            created_by TEXT NOT NULL, created_at TEXT NOT NULL,
            UNIQUE(asset_id,version_no)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_evidence_links (
            id {pk}, asset_id INTEGER NOT NULL, program_id INTEGER NOT NULL,
            cycle_id INTEGER, program_outcome_id INTEGER, improvement_action_id INTEGER,
            standard_code TEXT, criterion_code TEXT, linked_by TEXT NOT NULL, linked_at TEXT NOT NULL,
            UNIQUE(asset_id,program_id)
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_accreditation_portfolios (
            id {pk}, program_id INTEGER NOT NULL, title TEXT NOT NULL,
            accreditor TEXT, review_period TEXT, description TEXT,
            status TEXT NOT NULL DEFAULT 'draft', created_by TEXT NOT NULL,
            created_at TEXT NOT NULL, updated_at TEXT NOT NULL,
            frozen_at TEXT, frozen_by TEXT
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_accreditation_portfolio_items (
            id {pk}, portfolio_id INTEGER NOT NULL, asset_id INTEGER NOT NULL,
            evidence_version_id INTEGER NOT NULL, narrative TEXT,
            position INTEGER NOT NULL DEFAULT 1, added_by TEXT NOT NULL, added_at TEXT NOT NULL,
            UNIQUE(portfolio_id,asset_id)
        )""")


def _safe_url(value: str) -> str:
    clean = value.strip()
    if not clean:
        return ""
    parsed = urlparse(clean)
    if parsed.scheme not in {"http", "https"} or not parsed.netloc:
        raise HTTPException(400, "Evidence URL must use http or https.")
    return clean


def _program_asset(conn: Any, program_id: int, asset_id: int) -> dict[str, Any]:
    found = rows(execute(conn, """SELECT a.*,l.standard_code,l.criterion_code,l.cycle_id,l.program_outcome_id,l.improvement_action_id
        FROM nuvedra_evidence_assets a JOIN nuvedra_evidence_links l ON l.asset_id=a.id
        WHERE a.id=? AND l.program_id=? AND a.status='active'""", (asset_id, program_id)))
    if not found:
        raise HTTPException(404, "Evidence item not found in this program repository.")
    return found[0]


def _version(conn: Any, program_id: int, version_id: int) -> dict[str, Any]:
    found = rows(execute(conn, """SELECT v.*,a.title,a.evidence_type
        FROM nuvedra_evidence_versions v
        JOIN nuvedra_evidence_assets a ON a.id=v.asset_id
        JOIN nuvedra_evidence_links l ON l.asset_id=a.id
        WHERE v.id=? AND l.program_id=? AND a.status='active'""", (version_id, program_id)))
    if not found:
        raise HTTPException(404, "Evidence version not found.")
    return found[0]


def _latest_version(conn: Any, asset_id: int) -> dict[str, Any]:
    found = rows(execute(conn, "SELECT * FROM nuvedra_evidence_versions WHERE asset_id=? ORDER BY version_no DESC,id DESC LIMIT 1", (asset_id,)))
    if not found:
        raise HTTPException(409, "Evidence item has no stored version.")
    return found[0]


def _portfolio(conn: Any, program_id: int, portfolio_id: int) -> dict[str, Any]:
    found = rows(execute(conn, "SELECT * FROM nuvedra_accreditation_portfolios WHERE id=? AND program_id=?", (portfolio_id, program_id)))
    if not found:
        raise HTTPException(404, "Accreditation portfolio not found.")
    return found[0]


def _require_draft(portfolio: dict[str, Any]) -> None:
    if str(portfolio.get("status")) != "draft":
        raise HTTPException(409, "This accreditation portfolio is frozen and read-only.")


def _validate_context(conn: Any, program_id: int, cycle_id: int | None, outcome_id: int | None, action_id: int | None) -> None:
    if cycle_id is not None:
        if not rows(execute(conn, "SELECT id FROM nuvedra_assessment_cycles WHERE id=? AND program_id=?", (cycle_id, program_id))):
            raise HTTPException(400, "Assessment cycle does not belong to this program.")
    if outcome_id is not None:
        if not rows(execute(conn, "SELECT id FROM nuvedra_program_outcomes WHERE id=? AND program_id=?", (outcome_id, program_id))):
            raise HTTPException(400, "Program outcome does not belong to this program.")
    if action_id is not None:
        if not rows(execute(conn, """SELECT a.id FROM nuvedra_improvement_actions a
            JOIN nuvedra_assessment_cycles c ON c.id=a.cycle_id
            WHERE a.id=? AND c.program_id=?""", (action_id, program_id))):
            raise HTTPException(400, "Improvement action does not belong to this program.")


def _context_options(conn: Any, program_id: int) -> tuple[list[dict[str, Any]], list[dict[str, Any]], list[dict[str, Any]]]:
    cycles = rows(execute(conn, "SELECT id,label,status FROM nuvedra_assessment_cycles WHERE program_id=? ORDER BY id DESC", (program_id,)))
    outcomes = rows(execute(conn, "SELECT id,code,title FROM nuvedra_program_outcomes WHERE program_id=? AND status='active' ORDER BY code,title,id", (program_id,)))
    actions = rows(execute(conn, """SELECT a.id,a.title,a.status,c.label AS cycle_label FROM nuvedra_improvement_actions a
        JOIN nuvedra_assessment_cycles c ON c.id=a.cycle_id WHERE c.program_id=? ORDER BY a.id DESC""", (program_id,)))
    return cycles, outcomes, actions


def _program_assets(conn: Any, program_id: int) -> list[dict[str, Any]]:
    assets = rows(execute(conn, """SELECT a.*,l.standard_code,l.criterion_code,l.cycle_id,l.program_outcome_id,l.improvement_action_id
        FROM nuvedra_evidence_assets a JOIN nuvedra_evidence_links l ON l.asset_id=a.id
        WHERE l.program_id=? AND a.status='active' ORDER BY COALESCE(l.standard_code,''),COALESCE(l.criterion_code,''),a.title,a.id""", (program_id,)))
    for asset in assets:
        latest = _latest_version(conn, int(asset["id"]))
        asset["latest_version_id"] = int(latest["id"])
        asset["latest_version_no"] = int(latest["version_no"])
        asset["latest_file_name"] = latest.get("file_name")
        asset["latest_source_url"] = latest.get("source_url")
    return assets


def _portfolio_items(conn: Any, portfolio_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT pi.*,a.title,a.description,a.evidence_type,a.tags,
        v.version_no,v.file_name,v.mime_type,v.file_size,v.source_url,
        l.standard_code,l.criterion_code,l.cycle_id,l.program_outcome_id,l.improvement_action_id
        FROM nuvedra_accreditation_portfolio_items pi
        JOIN nuvedra_evidence_assets a ON a.id=pi.asset_id
        JOIN nuvedra_evidence_versions v ON v.id=pi.evidence_version_id
        JOIN nuvedra_evidence_links l ON l.asset_id=a.id
        JOIN nuvedra_accreditation_portfolios p ON p.id=pi.portfolio_id AND p.program_id=l.program_id
        WHERE pi.portfolio_id=? ORDER BY COALESCE(l.standard_code,''),COALESCE(l.criterion_code,''),pi.position,pi.id""", (portfolio_id,)))


def _ref_label(conn: Any, asset: dict[str, Any]) -> str:
    parts = []
    if asset.get("cycle_id"):
        found = rows(execute(conn, "SELECT label FROM nuvedra_assessment_cycles WHERE id=?", (int(asset["cycle_id"]),)))
        if found: parts.append(f"Cycle: {found[0].get('label')}")
    if asset.get("program_outcome_id"):
        found = rows(execute(conn, "SELECT code,title FROM nuvedra_program_outcomes WHERE id=?", (int(asset["program_outcome_id"]),)))
        if found: parts.append(f"Outcome: {found[0].get('code')} {found[0].get('title')}")
    if asset.get("improvement_action_id"):
        found = rows(execute(conn, "SELECT title FROM nuvedra_improvement_actions WHERE id=?", (int(asset["improvement_action_id"]),)))
        if found: parts.append(f"Action: {found[0].get('title')}")
    return " · ".join(parts) or "Program-level evidence"

    @app.post("/faculty/programs/{program_id}/evidence/assets", response_model=None)
    async def create_evidence(program_id: int, request: Request, title: str = Form(...), evidence_type: str = Form(...), description: str = Form(""), tags: str = Form(""), standard_code: str = Form(""), criterion_code: str = Form(""), cycle_id: str = Form(""), program_outcome_id: str = Form(""), improvement_action_id: str = Form(""), source_url: str = Form(""), change_note: str = Form("Initial evidence version"), upload: UploadFile | None = File(None)):
        user = _user(request, f"/faculty/programs/{program_id}/evidence")
        if isinstance(user, RedirectResponse): return user
        clean_title = title.strip(); kind = evidence_type.strip().lower()
        if not clean_title: raise HTTPException(400, "Evidence title is required.")
        if kind not in EVIDENCE_TYPES: raise HTTPException(400, "Unsupported evidence type.")
        url = _safe_url(source_url)
        data: bytes | None = None; file_name = ""; mime_type = ""
        if upload and upload.filename:
            file_name = PurePath(upload.filename).name
            if PurePath(file_name).suffix.lower() not in ALLOWED_EXTENSIONS:
                raise HTTPException(400, "This file extension is not allowed in the institutional evidence repository.")
            data = await upload.read(MAX_UPLOAD_BYTES + 1)
            if len(data) > MAX_UPLOAD_BYTES: raise HTTPException(413, "The uploaded evidence exceeds the 25 MB limit.")
            mime_type = str(upload.content_type or "application/octet-stream")
        if data is None and not url: raise HTTPException(400, "Upload an evidence file or provide an external URL.")
        cycle = int(cycle_id) if cycle_id.isdigit() else None
        outcome = int(program_outcome_id) if program_outcome_id.isdigit() else None
        action = int(improvement_action_id) if improvement_action_id.isdigit() else None
        with db() as conn:
            program_outcomes._require_manage(conn, request, program_id, user["email"])
            _validate_context(conn, program_id, cycle, outcome, action)
            now = utcnow()
            asset_id = _insert_id(conn, "INSERT INTO nuvedra_evidence_assets (title,description,evidence_type,tags,status,created_by,created_at,updated_at) VALUES (?,?,?,?,'active',?,?,?)", (clean_title, description.strip(), kind, tags.strip(), user["email"], now, now))
            _insert_id(conn, "INSERT INTO nuvedra_evidence_versions (asset_id,version_no,file_name,mime_type,file_size,file_bytes,source_url,change_note,created_by,created_at) VALUES (?,1,?,?,?,?,?,?,?,?)", (asset_id, file_name or None, mime_type or None, len(data or b""), data, url or None, change_note.strip(), user["email"], now))
            execute(conn, "INSERT INTO nuvedra_evidence_links (asset_id,program_id,cycle_id,program_outcome_id,improvement_action_id,standard_code,criterion_code,linked_by,linked_at) VALUES (?,?,?,?,?,?,?,?,?)", (asset_id, program_id, cycle, outcome, action, standard_code.strip(), criterion_code.strip(), user["email"], now))
            audit(conn, user["email"], "institutional_evidence_created", "evidence_asset", str(asset_id), f"program={program_id}; type={kind}", request.client.host if request.client else "")
        return RedirectResponse(f"/faculty/programs/{program_id}/evidence/assets/{asset_id}", status_code=303)

            program_outcomes._program_access(conn, request, program_id, user["email"])
            version = _version(conn, program_id, version_id)
        if version.get("file_bytes") is not None:
            filename = str(version.get("file_name") or f"evidence-v{version.get('version_no')}")
            return Response(content=bytes(version["file_bytes"]), media_type=str(version.get("mime_type") or "application/octet-stream"), headers={"Content-Disposition": f"inline; filename*=UTF-8''{quote(filename)}"})
        url = str(version.get("source_url") or "")
        if not url: raise HTTPException(404, "Evidence version has no file or URL.")
        return RedirectResponse(url, status_code=307)
