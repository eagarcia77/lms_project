STUDIO_PREFIX = "/faculty/studio"
TEAM_ID_RE = re.compile(r"^[A-Za-z0-9._-]{8,160}$")
REQUIRED_EDUCATION_SCOPES = {
    "User.ReadBasic.All",
    "Team.ReadBasic.All",
    "TeamMember.ReadWrite.All",
    "Channel.ReadBasic.All",
    "Channel.Create",
}


def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _insert_id(conn: Any, sql: str, params: tuple[Any, ...]) -> int:
    if database_url().startswith("postgres"):
        row = execute(conn, sql + " RETURNING id", params).fetchone()
        return int(row[0])
    return int(execute(conn, sql, params).lastrowid)


def ensure_microsoft365_education_schema() -> None:
    pk = _pk()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_microsoft_course_teams (
            id {pk}, course_id INTEGER NOT NULL UNIQUE, team_id TEXT NOT NULL,
            display_name TEXT NOT NULL, web_url TEXT, status TEXT NOT NULL DEFAULT 'active',
            linked_by TEXT NOT NULL, linked_at TEXT NOT NULL,
            last_synced_at TEXT, last_sync_json TEXT
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_microsoft_team_sync_log (
            id {pk}, course_id INTEGER NOT NULL, team_id TEXT NOT NULL,
            actor_email TEXT NOT NULL, action TEXT NOT NULL,
            summary_json TEXT, created_at TEXT NOT NULL
        )""")


def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def _admin(request: Request) -> dict[str, Any]:
    user = admin_console.session_user(request)
    if not user or str(user.get("role") or "") not in {"superadmin", "course_admin"}:
        raise HTTPException(403, "Platform administration permission is required.")
    return user


def _team_id(value: str) -> str:
    clean = str(value or "").strip()
    if not TEAM_ID_RE.fullmatch(clean):
        raise HTTPException(400, "Microsoft Team identifier is invalid.")
    return clean


def _team_link(conn: Any, course_id: int) -> dict[str, Any] | None:
    found = rows(execute(conn, "SELECT * FROM nuvedra_microsoft_course_teams WHERE course_id=? AND status='active'", (course_id,)))
    return found[0] if found else None


def _configured_scopes() -> set[str]:
    raw = str(os.getenv("MICROSOFT_SCOPES", m365.DEFAULT_SCOPES) or "")
    return {part.strip() for part in raw.split() if part.strip()}


def _scope_report() -> tuple[list[str], list[str]]:
    configured = _configured_scopes()
    present = sorted(REQUIRED_EDUCATION_SCOPES.intersection(configured))
    missing = sorted(REQUIRED_EDUCATION_SCOPES.difference(configured))
    return present, missing


def _course_members(conn: Any, course_id: int) -> list[dict[str, Any]]:
    allowed = set(academic_access.AUTHOR_ROLES) | {"student"}
    data = rows(execute(conn, """SELECT lower(user_email) AS email,course_role
        FROM nexus_admin_enrollments WHERE course_id=? AND status='active'
        ORDER BY course_role,lower(user_email)""", (course_id,)))
    return [row for row in data if str(row.get("course_role") or "") in allowed]


def _member_email(member: dict[str, Any]) -> str:
    return str(member.get("email") or member.get("userPrincipalName") or "").strip().lower()


def _target_owner(role: str) -> bool:
    return role == "instructor"


async def _team_members(email: str, team_id: str) -> list[dict[str, Any]]:
    payload = await m365._graph_json(email, "GET", f"/teams/{_team_id(team_id)}/members", params={"$top": 200})
    if payload.get("@odata.nextLink"):
        raise HTTPException(409, "This Team has more than 200 members. Membership synchronization is paused until paginated Graph membership support is enabled.")
    values = payload.get("value", [])
    return values if isinstance(values, list) else []


async def _resolve_graph_user(email: str, target_email: str) -> dict[str, Any]:
    encoded = quote(target_email.strip().lower(), safe="")
    return await m365._graph_json(email, "GET", f"/users/{encoded}", params={"$select": "id,displayName,mail,userPrincipalName"})


def _member_payload(user_id: str, owner: bool) -> dict[str, Any]:
    return {
        "@odata.type": "#microsoft.graph.aadUserConversationMember",
        "roles": ["owner"] if owner else [],
        "user@odata.bind": f"https://graph.microsoft.com/v1.0/users('{user_id}')",
    }


async def _membership_plan(conn: Any, actor_email: str, course_id: int, team: dict[str, Any]) -> dict[str, Any]:
    target = _course_members(conn, course_id)
    current = await _team_members(actor_email, str(team["team_id"]))
    by_email = {_member_email(member): member for member in current if _member_email(member)}
    missing: list[dict[str, Any]] = []
    matched: list[dict[str, Any]] = []
    role_mismatches: list[dict[str, Any]] = []
    for row in target:
        email = str(row["email"]).lower()
        role = str(row["course_role"])
        member = by_email.get(email)
        if not member:
            missing.append({"email": email, "course_role": role, "teams_role": "owner" if _target_owner(role) else "member"})
            continue
        roles = {str(x).lower() for x in (member.get("roles") or [])}
        matched.append({"email": email, "course_role": role, "teams_roles": sorted(roles)})
        if _target_owner(role) and "owner" not in roles:
            role_mismatches.append({"email": email, "expected": "owner", "actual": sorted(roles) or ["member"]})
    unmanaged = [
        {"email": _member_email(member) or "(no email exposed)", "display_name": member.get("displayName")}
        for member in current
        if not _member_email(member) or _member_email(member) not in {str(x["email"]).lower() for x in target}
    ]
    return {
        "target_count": len(target),
        "team_member_count": len(current),
        "missing": missing,
        "matched": matched,
        "role_mismatches": role_mismatches,
        "unmanaged": unmanaged,
        "sync_mode": "additive",
    }


def _record_sync(conn: Any, course_id: int, team_id: str, actor_email: str, action: str, summary: dict[str, Any]) -> None:
    now = utcnow()
    encoded = json.dumps(summary, ensure_ascii=False)
    execute(conn, "INSERT INTO nuvedra_microsoft_team_sync_log (course_id,team_id,actor_email,action,summary_json,created_at) VALUES (?,?,?,?,?,?)",
            (course_id, team_id, actor_email.lower(), action, encoded, now))
    execute(conn, "UPDATE nuvedra_microsoft_course_teams SET last_synced_at=?,last_sync_json=? WHERE course_id=? AND status='active'",
            (now, encoded, course_id))

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/microsoft365/education/team", response_model=None)
    async def link_team(course_id: int, request: Request, team_id: str = Form(...)):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/education")
        if isinstance(user, RedirectResponse): return user
        clean_id = _team_id(team_id)
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
        team = await m365._graph_json(user["email"], "GET", f"/teams/{clean_id}", params={"$select": "id,displayName,webUrl"})
        name = str(team.get("displayName") or "Microsoft Team").strip()
        web_url = str(team.get("webUrl") or "").strip()
        now = utcnow()
        with db() as conn:
            existing = _team_link(conn, course_id)
            if existing:
                execute(conn, "UPDATE nuvedra_microsoft_course_teams SET team_id=?,display_name=?,web_url=?,linked_by=?,linked_at=?,last_synced_at=NULL,last_sync_json=NULL WHERE course_id=?",
                        (clean_id, name, web_url or None, user["email"], now, course_id))
            else:
                execute(conn, "INSERT INTO nuvedra_microsoft_course_teams (course_id,team_id,display_name,web_url,status,linked_by,linked_at,last_synced_at,last_sync_json) VALUES (?,?,?,?,'active',?,?,NULL,NULL)",
                        (course_id, clean_id, name, web_url or None, user["email"], now))
            audit(conn, user["email"], "microsoft365_team_linked", "course", str(course_id), clean_id, request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/education", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/microsoft365/education/sync", response_model=None)
    async def sync_members(course_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/education")
        if isinstance(user, RedirectResponse): return user
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            team = _team_link(conn, course_id)
            if not team: raise HTTPException(409, "Link a Microsoft Team before synchronizing membership.")
            plan = await _membership_plan(conn, user["email"], course_id, team)
        added: list[dict[str, Any]] = []
        unresolved: list[dict[str, Any]] = []
        for target in plan["missing"]:
            target_email = str(target["email"])
            try:
                graph_user = await _resolve_graph_user(user["email"], target_email)
                user_id = str(graph_user.get("id") or "").strip()
                if not user_id:
                    raise HTTPException(404, "Microsoft user ID was not returned.")
                await m365._graph_json(user["email"], "POST", f"/teams/{_team_id(str(team['team_id']))}/members", json_body=_member_payload(user_id, target["teams_role"] == "owner"))
                added.append({"email": target_email, "teams_role": target["teams_role"], "user_id": user_id})
            except HTTPException as exc:
                unresolved.append({"email": target_email, "status": exc.status_code, "detail": str(exc.detail)[:300]})
        summary = {
            "mode": "additive",
            "target_count": plan["target_count"],
            "added": added,
            "unresolved": unresolved,
            "role_mismatches_preserved": plan["role_mismatches"],
            "unmanaged_preserved": len(plan["unmanaged"]),
        }
        with db() as conn:
            _record_sync(conn, course_id, str(team["team_id"]), user["email"], "membership_additive_sync", summary)
            audit(conn, user["email"], "microsoft365_team_membership_synced", "course", str(course_id), json.dumps({"added": len(added), "unresolved": len(unresolved)}), request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/education", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/microsoft365/education/channels", response_model=None)
    async def create_channel(course_id: int, request: Request, display_name: str = Form(...), description: str = Form("")):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/education")
        if isinstance(user, RedirectResponse): return user
        name = display_name.strip()
        if not name or len(name) > 50: raise HTTPException(400, "Channel name is required and must be 50 characters or fewer.")
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            team = _team_link(conn, course_id)
            if not team: raise HTTPException(409, "Link a Microsoft Team before creating channels.")
        channel = await m365._graph_json(user["email"], "POST", f"/teams/{_team_id(str(team['team_id']))}/channels", json_body={"displayName": name, "description": description.strip()[:500], "membershipType": "standard"})
        summary = {"channel_id": channel.get("id"), "display_name": channel.get("displayName") or name}
        with db() as conn:
            _record_sync(conn, course_id, str(team["team_id"]), user["email"], "channel_created", summary)
            audit(conn, user["email"], "microsoft365_team_channel_created", "course", str(course_id), name, request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/education", status_code=303)

    async def student_microsoft365(course_id: int, request: Request):
