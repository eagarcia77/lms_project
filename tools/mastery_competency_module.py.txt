STUDIO_PREFIX = "/faculty/studio"


def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def ensure_mastery_competency_schema() -> None:
    pk = _pk()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_mastery_settings (
            id {pk}, course_id INTEGER NOT NULL UNIQUE,
            proficient_threshold REAL NOT NULL DEFAULT 80,
            mastery_threshold REAL NOT NULL DEFAULT 90,
            updated_by TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")


def _settings(conn: Any, course_id: int) -> dict[str, Any]:
    found = rows(execute(conn, "SELECT * FROM nuvedra_mastery_settings WHERE course_id=?", (course_id,)))
    if found:
        return found[0]
    return {"course_id": course_id, "proficient_threshold": 80.0, "mastery_threshold": 90.0}


def _outcomes(conn: Any, course_id: int) -> list[dict[str, Any]]:
    data = rows(execute(conn, """SELECT o.*,
        (SELECT COUNT(*) FROM nuvedra_outcome_links ol WHERE ol.outcome_id=o.id) AS aligned_items
        FROM nuvedra_outcomes o
        WHERE o.course_id=? AND o.status='active'
        ORDER BY o.code,o.title,o.id""", (course_id,)))
    return data


def _students(conn: Any, course_id: int) -> list[str]:
    data = rows(execute(conn, """SELECT DISTINCT lower(user_email) AS email
        FROM nexus_admin_enrollments
        WHERE course_id=? AND status='active' AND course_role='student'
        ORDER BY lower(user_email)""", (course_id,)))
    return [str(row.get("email") or "").lower() for row in data if str(row.get("email") or "").strip()]


def _evidence(conn: Any, outcome_id: int, email: str) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT i.id AS item_id,i.title AS item_title,i.item_type,i.points,
        s.id AS submission_id,s.status AS submission_status,
        g.points_awarded,g.status AS grade_status,g.graded_at,
        re.raw_score AS rubric_raw_score,re.possible_raw_score AS rubric_possible_raw_score,
        re.grade_points AS rubric_grade_points,re.graded_at AS rubric_graded_at
        FROM nuvedra_outcome_links ol
        JOIN nexus_content_items i ON i.id=ol.item_id
        LEFT JOIN nuvedra_submissions s ON s.item_id=i.id AND lower(s.student_email)=?
        LEFT JOIN nuvedra_grades g ON g.submission_id=s.id
        LEFT JOIN nuvedra_rubric_links rl ON rl.item_id=i.id
        LEFT JOIN nuvedra_rubric_evaluations re ON re.submission_id=s.id AND re.rubric_id=rl.rubric_id AND re.status='graded'
        WHERE ol.outcome_id=?
        ORDER BY i.title,i.id""", (email.lower(), outcome_id)))


def _attainment(evidence: list[dict[str, Any]]) -> tuple[float | None, int, int, int]:
    earned = 0.0
    possible = 0.0
    graded = 0
    rubric_evidence = 0
    for row in evidence:
        item_points = float(row.get("points") or 0)
        if str(row.get("grade_status") or "") == "graded" and row.get("points_awarded") is not None and item_points > 0:
            awarded = max(0.0, min(item_points, float(row.get("points_awarded") or 0)))
            earned += awarded
            possible += item_points
            graded += 1
        if row.get("rubric_raw_score") is not None and float(row.get("rubric_possible_raw_score") or 0) > 0:
            rubric_evidence += 1
    return (None if possible <= 0 else (earned / possible) * 100.0), graded, len(evidence), rubric_evidence


def _level(percent: float | None, settings: dict[str, Any]) -> dict[str, str]:
    proficient = float(settings.get("proficient_threshold") or 80)
    mastery = float(settings.get("mastery_threshold") or 90)
    if percent is None:
        return {"key": "no-evidence", "en": "No evidence", "es": "Sin evidencia"}
    if percent + 1e-9 >= mastery:
        return {"key": "mastered", "en": "Mastered", "es": "Dominado"}
    if percent + 1e-9 >= proficient:
        return {"key": "proficient", "en": "Proficient", "es": "Competente"}
    return {"key": "developing", "en": "Developing", "es": "En desarrollo"}


def _outcome_label(outcome: dict[str, Any]) -> str:
    code = str(outcome.get("code") or "").strip()
    title = str(outcome.get("title") or "Learning outcome")
    return f"{code}: {title}" if code else title


def _student_mastery(conn: Any, course_id: int, email: str, settings: dict[str, Any], outcomes: list[dict[str, Any]] | None = None) -> list[dict[str, Any]]:
    outcome_rows = outcomes if outcomes is not None else _outcomes(conn, course_id)
    result: list[dict[str, Any]] = []
    for outcome in outcome_rows:
        evidence = _evidence(conn, int(outcome["id"]), email)
        percent, graded, aligned, rubric_evidence = _attainment(evidence)
        result.append({
            "outcome": outcome,
            "evidence": evidence,
            "percent": percent,
            "graded": graded,
            "aligned": aligned,
            "rubric_evidence": rubric_evidence,
            "level": _level(percent, settings),
        })
    return result


def _evidence_table(evidence: list[dict[str, Any]]) -> str:
    body = []
    for row in evidence:
        possible = float(row.get("points") or 0)
        if str(row.get("grade_status") or "") == "graded" and row.get("points_awarded") is not None and possible > 0:
            awarded = float(row.get("points_awarded") or 0)
            grade = f"{awarded:g}/{possible:g} · {(awarded / possible) * 100:.1f}%"
        else:
            grade = "Awaiting graded evidence"
        rubric = "—"
        rubric_possible = float(row.get("rubric_possible_raw_score") or 0)
        if row.get("rubric_raw_score") is not None and rubric_possible > 0:
            rubric = f"{float(row.get('rubric_raw_score') or 0):g}/{rubric_possible:g} rubric points"
        body.append(f'''<tr><td>{academic_access.esc(row.get('item_title'))}<br><small>{academic_access.esc(row.get('item_type'))}</small></td><td>{academic_access.esc(grade)}</td><td>{academic_access.esc(rubric)}</td></tr>''')
    return "".join(body) or '<tr><td colspan="3" data-i18n-en="No activities are aligned to this outcome yet." data-i18n-es="Todavía no hay actividades alineadas con este resultado.">No activities are aligned to this outcome yet.</td></tr>'


def _student_cards(data: list[dict[str, Any]]) -> str:
    cards = []
    for entry in data:
        outcome = entry["outcome"]
        percent = entry["percent"]
        level = entry["level"]
        percent_text = "—" if percent is None else f"{percent:.1f}%"
        cards.append(f'''<article class="studio-panel" data-mastery-level="{academic_access.esc(level['key'], attr=True)}"><p class="studio-eyebrow">{academic_access.esc(outcome.get('code') or 'Outcome')}</p><h3>{academic_access.esc(outcome.get('title'))}</h3><p>{academic_access.esc(outcome.get('description'))}</p><p><strong>{academic_access.esc(percent_text)}</strong> · <span data-i18n-en="{academic_access.esc(level['en'], attr=True)}" data-i18n-es="{academic_access.esc(level['es'], attr=True)}">{academic_access.esc(level['en'])}</span> · {int(entry['graded'])}/{int(entry['aligned'])} graded evidence items</p><details><summary data-i18n-en="View evidence" data-i18n-es="Ver evidencia">View evidence</summary><div style="overflow:auto"><table><thead><tr><th data-i18n-en="Evidence" data-i18n-es="Evidencia">Evidence</th><th data-i18n-en="Grade" data-i18n-es="Calificación">Grade</th><th data-i18n-en="Rubric evidence" data-i18n-es="Evidencia de rúbrica">Rubric evidence</th></tr></thead><tbody>{_evidence_table(entry['evidence'])}</tbody></table></div></details></article>''')
    return "".join(cards) or '<section class="studio-empty" data-i18n-en="No active learning outcomes are configured for this course." data-i18n-es="No hay resultados de aprendizaje activos configurados para este curso.">No active learning outcomes are configured for this course.</section>'

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/mastery/settings", response_model=None)
    async def save_mastery_settings(course_id: int, request: Request, proficient_threshold: float = Form(80), mastery_threshold: float = Form(90)):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/mastery")
        if isinstance(user, RedirectResponse):
            return user
        if proficient_threshold < 0 or proficient_threshold > 100 or mastery_threshold < 0 or mastery_threshold > 100:
            raise HTTPException(400, "Mastery thresholds must be between 0 and 100.")
        if mastery_threshold + 1e-9 < proficient_threshold:
            raise HTTPException(400, "The mastery threshold cannot be lower than the proficiency threshold.")
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            now = utcnow()
            found = rows(execute(conn, "SELECT id FROM nuvedra_mastery_settings WHERE course_id=?", (course_id,)))
            if found:
                execute(conn, "UPDATE nuvedra_mastery_settings SET proficient_threshold=?,mastery_threshold=?,updated_by=?,updated_at=? WHERE course_id=?", (proficient_threshold, mastery_threshold, user["email"], now, course_id))
            else:
                execute(conn, "INSERT INTO nuvedra_mastery_settings (course_id,proficient_threshold,mastery_threshold,updated_by,updated_at) VALUES (?,?,?,?,?)", (course_id, proficient_threshold, mastery_threshold, user["email"], now))
            audit(conn, user["email"], "mastery_thresholds_updated", "course", str(course_id), f"proficient={proficient_threshold:g};mastered={mastery_threshold:g}", request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{course_id}/mastery", status_code=303)

    @app.get(f"{STUDIO_PREFIX}/courses/{{course_id}}/mastery.csv", response_model=None)
    async def export_mastery(course_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/mastery")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            course = academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            settings = _settings(conn, course_id)
            outcomes = _outcomes(conn, course_id)
            students = _students(conn, course_id)
            payload = [(email, _student_mastery(conn, course_id, email, settings, outcomes)) for email in students]
        stream = io.StringIO()
        writer = csv.writer(stream)
        writer.writerow(["Student", "Outcome code", "Outcome title", "Attainment percent", "Level", "Graded evidence", "Aligned activities", "Rubric evidence"])
        for email, data in payload:
            for entry in data:
                outcome = entry["outcome"]
                writer.writerow([email, outcome.get("code") or "", outcome.get("title") or "", "" if entry["percent"] is None else round(float(entry["percent"]), 2), entry["level"]["en"], entry["graded"], entry["aligned"], entry["rubric_evidence"]])
        filename = f"{str(course.get('course_code') or 'course').replace(' ', '_')}-mastery-competencies.csv"
        return Response(stream.getvalue(), media_type="text/csv; charset=utf-8", headers={"Content-Disposition": f'attachment; filename="{filename}"'})
