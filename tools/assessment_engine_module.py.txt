ASSESSMENT_TYPES_V2 = {"assessment", "quiz"}
QUESTION_TYPES = {"multiple_choice", "true_false", "short_answer", "essay"}
STUDIO_PREFIX = "/faculty/studio"


def _esc(value: Any, *, attr: bool = False) -> str:
    return html.escape(str(value or ""), quote=attr)


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def ensure_assessment_schema() -> None:
    pk = _pk()
    with db() as conn:
        execute(
            conn,
            f"""CREATE TABLE IF NOT EXISTS nuvedra_assessment_questions (
                id {pk},
                item_id INTEGER NOT NULL,
                question_type TEXT NOT NULL,
                prompt TEXT NOT NULL,
                choices_json TEXT NOT NULL DEFAULT '[]',
                correct_answer TEXT,
                points REAL NOT NULL DEFAULT 1,
                position INTEGER NOT NULL DEFAULT 1,
                feedback_correct TEXT,
                feedback_incorrect TEXT,
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL
            )""",
        )
        execute(
            conn,
            f"""CREATE TABLE IF NOT EXISTS nuvedra_question_bank (
                id {pk},
                owner_email TEXT NOT NULL,
                question_type TEXT NOT NULL,
                prompt TEXT NOT NULL,
                choices_json TEXT NOT NULL DEFAULT '[]',
                correct_answer TEXT,
                points REAL NOT NULL DEFAULT 1,
                tags TEXT,
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL
            )""",
        )
        execute(
            conn,
            f"""CREATE TABLE IF NOT EXISTS nuvedra_assessment_attempts (
                id {pk},
                item_id INTEGER NOT NULL,
                student_email TEXT NOT NULL,
                attempt_number INTEGER NOT NULL,
                status TEXT NOT NULL DEFAULT 'in_progress',
                started_at TEXT NOT NULL,
                submitted_at TEXT,
                score_auto REAL NOT NULL DEFAULT 0,
                score_total REAL,
                has_manual_items INTEGER NOT NULL DEFAULT 0,
                UNIQUE(item_id,student_email,attempt_number)
            )""",
        )
        execute(
            conn,
            f"""CREATE TABLE IF NOT EXISTS nuvedra_assessment_answers (
                id {pk},
                attempt_id INTEGER NOT NULL,
                question_id INTEGER NOT NULL,
                answer_text TEXT,
                is_correct INTEGER,
                points_awarded REAL,
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL,
                UNIQUE(attempt_id,question_id)
            )""",
        )


def _insert_id(conn: Any, sql: str, params: tuple[Any, ...]) -> int:
    if database_url().startswith("postgres"):
        row = execute(conn, sql + " RETURNING id", params).fetchone()
        return int(row[0])
    cursor = execute(conn, sql, params)
    return int(cursor.lastrowid)


def _metadata(item: dict[str, Any]) -> dict[str, Any]:
    try:
        value = json.loads(str(item.get("metadata_json") or "{}"))
    except json.JSONDecodeError:
        return {}
    return value if isinstance(value, dict) else {}


def _assessment_settings(item: dict[str, Any]) -> dict[str, Any]:
    meta = _metadata(item)
    settings = meta.get("assessment")
    if not isinstance(settings, dict):
        settings = {}
    attempts = max(1, int(settings.get("attempts") or 1))
    time_limit = max(0, int(settings.get("time_limit") or 0))
    return {
        "attempts": attempts,
        "time_limit": time_limit,
        "response_type": str(settings.get("response_type") or "structured"),
        "rubric": str(settings.get("rubric") or ""),
    }


def _item_bundle(conn: Any, item_id: int) -> tuple[int, dict[str, Any], dict[str, Any]]:
    course_id, item, module = academic_access.item_bundle(conn, item_id)
    if str(item.get("item_type")) not in ASSESSMENT_TYPES_V2:
        raise HTTPException(400, "This item is not a structured assessment.")
    return course_id, item, module


def _author(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def _author_access(conn: Any, item_id: int, user: dict[str, Any]) -> tuple[int, dict[str, Any], dict[str, Any]]:
    course_id, item, module = _item_bundle(conn, item_id)
    academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
    return course_id, item, module


def _questions(conn: Any, item_id: int) -> list[dict[str, Any]]:
    return rows(
        execute(
            conn,
            "SELECT * FROM nuvedra_assessment_questions WHERE item_id=? ORDER BY position,id",
            (item_id,),
        )
    )


def _normalize_question_type(value: str) -> str:
    value = value.strip()
    if value not in QUESTION_TYPES:
        raise HTTPException(400, "Invalid question type.")
    return value


def _choices(question_type: str, raw: str) -> list[str]:
    if question_type == "true_false":
        return ["True", "False"]
    if question_type != "multiple_choice":
        return []
    values = [line.strip() for line in raw.splitlines() if line.strip()]
    if len(values) < 2:
        raise HTTPException(400, "Multiple-choice questions require at least two options.")
    if len(set(values)) != len(values):
        raise HTTPException(400, "Multiple-choice options must be unique.")
    return values


def _validate_correct_answer(question_type: str, choices: list[str], value: str) -> str:
    answer = value.strip()
    if question_type == "essay":
        return ""
    if not answer:
        raise HTTPException(400, "A correct answer is required for automatically graded questions.")
    if question_type in {"multiple_choice", "true_false"} and answer not in choices:
        raise HTTPException(400, "The correct answer must match one of the available options.")
    return answer

def _student_access(conn: Any, item_id: int, user: dict[str, Any]) -> tuple[int, dict[str, Any], dict[str, Any], dict[str, Any]]:
    course_id, item, module = _item_bundle(conn, item_id)
    access = academic_access.require_course_role(conn, course_id, user["email"], {"student"})
    if (
        str(access.get("course_status")) != "active"
        or str(module.get("status")) != "published"
        or str(item.get("status")) != "published"
    ):
        raise HTTPException(403, "This assessment is not available.")
    return course_id, item, module, access


def _parse_iso(value: str) -> datetime:
    clean = str(value or "").strip().replace("Z", "+00:00")
    parsed = datetime.fromisoformat(clean)
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=timezone.utc)
    return parsed.astimezone(timezone.utc)


def _deadline_ms(started_at: str, minutes: int) -> int:
    if minutes <= 0:
        return 0
    start = _parse_iso(started_at)
    return int((start.timestamp() + minutes * 60) * 1000)


def _answer_input(question: dict[str, Any], existing: str = "") -> str:
    qid = int(question["id"])
    qtype = str(question.get("question_type"))
    name = f"q_{qid}"
    if qtype in {"multiple_choice", "true_false"}:
        try:
            choices = json.loads(str(question.get("choices_json") or "[]"))
        except json.JSONDecodeError:
            choices = []
        controls = []
        for option in choices:
            checked = " checked" if str(existing) == str(option) else ""
            controls.append(
                f'<label class="studio-choice"><input type="radio" name="{name}" value="{_esc(option, attr=True)}"{checked} required> {_esc(option)}</label>'
            )
        return "".join(controls)
    if qtype == "short_answer":
        return f'<input name="{name}" value="{_esc(existing, attr=True)}" required>'
    return f'<textarea name="{name}" rows="6" required>{_esc(existing)}</textarea>'


def _grade_answer(question: dict[str, Any], answer: str) -> tuple[int | None, float | None]:
    qtype = str(question.get("question_type"))
    if qtype == "essay":
        return None, None
    actual = answer.strip().casefold()
    expected = str(question.get("correct_answer") or "")
    if qtype == "short_answer":
        accepted = [part.strip().casefold() for part in expected.split("||") if part.strip()]
        correct = actual in accepted
    else:
        correct = actual == expected.strip().casefold()
    points = float(question.get("points") or 0) if correct else 0.0
    return (1 if correct else 0), points


def _upsert_submission(conn: Any, item_id: int, email: str, attempt_number: int, status: str) -> int:
    now = utcnow()
    summary = f"Structured assessment attempt {attempt_number} submitted ({status})."
    found = rows(execute(conn, "SELECT id FROM nuvedra_submissions WHERE item_id=? AND lower(student_email)=?", (item_id, email.lower())))
    if found:
        submission_id = int(found[0]["id"])
        execute(
            conn,
            "UPDATE nuvedra_submissions SET response_text=?,response_url=NULL,status='submitted',submitted_at=?,updated_at=? WHERE id=?",
            (summary, now, now, submission_id),
        )
        return submission_id
    return _insert_id(
        conn,
        "INSERT INTO nuvedra_submissions (item_id,student_email,response_text,response_url,status,submitted_at,updated_at) VALUES (?,?,?,?,?,?,?)",
        (item_id, email.lower(), summary, None, "submitted", now, now),
    )


def _sync_auto_grade(conn: Any, submission_id: int, score: float, user_email: str, fully_auto: bool) -> None:
    if not fully_auto:
        execute(conn, "DELETE FROM nuvedra_grades WHERE submission_id=?", (submission_id,))
        return
    now = utcnow()
    found = rows(execute(conn, "SELECT id FROM nuvedra_grades WHERE submission_id=?", (submission_id,)))
    feedback = "Automatically graded by NUVEDRA Assessments."
    if found:
        execute(
            conn,
            "UPDATE nuvedra_grades SET points_awarded=?,feedback=?,status='graded',graded_by=?,graded_at=?,updated_at=? WHERE submission_id=?",
            (score, feedback, "NUVEDRA AutoGrade", now, now, submission_id),
        )
    else:
        execute(
            conn,
            "INSERT INTO nuvedra_grades (submission_id,points_awarded,feedback,status,graded_by,graded_at,updated_at) VALUES (?,?,?,?,?,?,?)",
            (submission_id, score, feedback, "graded", "NUVEDRA AutoGrade", now, now),
        )


def register_assessment_engine(app: FastAPI) -> None:
    ensure_assessment_schema()

        user = _author(request, f"{STUDIO_PREFIX}/items/{item_id}/assessment")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            course_id, item, module = _author_access(conn, item_id, user)
            questions = _questions(conn, item_id)
            bank = rows(
                execute(
                    conn,
                    "SELECT * FROM nuvedra_question_bank WHERE lower(owner_email)=? ORDER BY updated_at DESC,id DESC LIMIT 50",
                    (user["email"].lower(),),
                )
            )
        settings = _assessment_settings(item)
        total_points = sum(float(question.get("points") or 0) for question in questions)
        question_html = "".join(_question_card(item_id, question) for question in questions) or '<section class="studio-empty" data-i18n-en="No questions yet. Add the first question below." data-i18n-es="Todavía no hay preguntas. Añada la primera pregunta abajo.">No questions yet. Add the first question below.</section>'
        bank_html = "".join(_bank_card(item_id, question) for question in bank) or '<section class="studio-empty" data-i18n-en="Your question bank is empty." data-i18n-es="Su banco de preguntas está vacío.">Your question bank is empty.</section>'
        body = f"""{_assets()}
        <main class="studio-shell" data-studio-root data-testid="assessment-builder">
          <nav class="studio-breadcrumbs"><a href="{STUDIO_PREFIX}/items/{item_id}/edit" data-i18n-en="Back to item" data-i18n-es="Volver al elemento">Back to item</a><span>/</span><a href="{STUDIO_PREFIX}/modules/{int(module['id'])}">{_esc(module.get('title'))}</a></nav>
          <header class="studio-hero studio-hero--module">
            <div><p class="studio-eyebrow" data-i18n-en="Assessment Builder" data-i18n-es="Constructor de evaluaciones">Assessment Builder</p><h2>{_esc(item.get('title'))}</h2><p data-i18n-en="Build structured questions without writing code." data-i18n-es="Cree preguntas estructuradas sin escribir código.">Build structured questions without writing code.</p></div>
            <div class="studio-actions"><a class="studio-button studio-button--quiet" href="{STUDIO_PREFIX}/courses/{course_id}/gradebook">Gradebook</a></div>
          </header>
          <section class="studio-summary-grid">
            <article class="studio-panel"><strong>{len(questions)}</strong><span data-i18n-en="Questions" data-i18n-es="Preguntas">Questions</span></article>
            <article class="studio-panel"><strong>{total_points:g}</strong><span data-i18n-en="Question points" data-i18n-es="Puntos de preguntas">Question points</span></article>
            <article class="studio-panel"><strong>{settings['attempts']}</strong><span data-i18n-en="Attempts" data-i18n-es="Intentos">Attempts</span></article>

          </section>
          <section class="studio-section"><h2 data-i18n-en="Assessment questions" data-i18n-es="Preguntas de la evaluación">Assessment questions</h2>{question_html}</section>

    @app.post(f"{STUDIO_PREFIX}/items/{{item_id}}/assessment/questions", response_model=None)
    async def add_question(
        item_id: int,
        request: Request,
        question_type: str = Form(...),
        prompt: str = Form(...),
        choices: str = Form(""),
        correct_answer: str = Form(""),
        points: float = Form(1),
        position: int = Form(1),
        feedback_correct: str = Form(""),
        feedback_incorrect: str = Form(""),
        save_to_bank: str = Form(""),
    ):
        user = _author(request, f"{STUDIO_PREFIX}/items/{item_id}/assessment")
        if isinstance(user, RedirectResponse):
            return user
        qtype = _normalize_question_type(question_type)
        clean_prompt = prompt.strip()
        if not clean_prompt:
            raise HTTPException(400, "Question text is required.")
        if points < 0:
            raise HTTPException(400, "Question points cannot be negative.")
        options = _choices(qtype, choices)
        answer = _validate_correct_answer(qtype, options, correct_answer)
        now = utcnow()
        with db() as conn:
            _author_access(conn, item_id, user)
            _insert_id(
                conn,
                """INSERT INTO nuvedra_assessment_questions
                   (item_id,question_type,prompt,choices_json,correct_answer,points,position,feedback_correct,feedback_incorrect,created_at,updated_at)
                   VALUES (?,?,?,?,?,?,?,?,?,?,?)""",
                (item_id, qtype, clean_prompt, json.dumps(options), answer or None, points, max(1, position), feedback_correct.strip(), feedback_incorrect.strip(), now, now),
            )
            if save_to_bank == "1":
                _insert_id(
                    conn,
                    """INSERT INTO nuvedra_question_bank
                       (owner_email,question_type,prompt,choices_json,correct_answer,points,tags,created_at,updated_at)
                       VALUES (?,?,?,?,?,?,?,?,?)""",
                    (user["email"].lower(), qtype, clean_prompt, json.dumps(options), answer or None, points, "", now, now),
                )
            audit(conn, user["email"], "assessment_question_created", "item", str(item_id), qtype, request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/items/{item_id}/assessment", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/items/{{item_id}}/assessment/questions/{{question_id}}/update", response_model=None)
    async def update_question(
        item_id: int,
        question_id: int,
        request: Request,
        question_type: str = Form(...),
        prompt: str = Form(...),
        choices: str = Form(""),
        correct_answer: str = Form(""),
        points: float = Form(1),
        position: int = Form(1),
        feedback_correct: str = Form(""),
        feedback_incorrect: str = Form(""),
    ):
        user = _author(request, f"{STUDIO_PREFIX}/items/{item_id}/assessment")
        if isinstance(user, RedirectResponse):
            return user
        qtype = _normalize_question_type(question_type)
        clean_prompt = prompt.strip()
        if not clean_prompt:
            raise HTTPException(400, "Question text is required.")
        if points < 0:
            raise HTTPException(400, "Question points cannot be negative.")
        options = _choices(qtype, choices)
        answer = _validate_correct_answer(qtype, options, correct_answer)
        with db() as conn:
            _author_access(conn, item_id, user)
            found = rows(execute(conn, "SELECT id FROM nuvedra_assessment_questions WHERE id=? AND item_id=?", (question_id, item_id)))
            if not found:
                raise HTTPException(404, "Question not found.")
            execute(
                conn,
                """UPDATE nuvedra_assessment_questions
                   SET question_type=?,prompt=?,choices_json=?,correct_answer=?,points=?,position=?,feedback_correct=?,feedback_incorrect=?,updated_at=?
                   WHERE id=? AND item_id=?""",
                (qtype, clean_prompt, json.dumps(options), answer or None, points, max(1, position), feedback_correct.strip(), feedback_incorrect.strip(), utcnow(), question_id, item_id),
            )
            audit(conn, user["email"], "assessment_question_updated", "question", str(question_id), qtype, request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/items/{item_id}/assessment", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/items/{{item_id}}/assessment/questions/{{question_id}}/delete", response_model=None)
    async def delete_question(item_id: int, question_id: int, request: Request):
        user = _author(request, f"{STUDIO_PREFIX}/items/{item_id}/assessment")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            _author_access(conn, item_id, user)
            execute(conn, "DELETE FROM nuvedra_assessment_questions WHERE id=? AND item_id=?", (question_id, item_id))
            audit(conn, user["email"], "assessment_question_deleted", "question", str(question_id), "", request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/items/{item_id}/assessment", status_code=303)

    @app.get("/learn/assessments/{item_id}", response_class=HTMLResponse, response_model=None)

            return academic_access.login_redirect(f"/learn/assessments/{item_id}")

            questions = _questions(conn, item_id)
            attempts = rows(
                execute(
                    conn,

        assessment_form = ""
        if active:
            with db() as conn:
                previous_answers = rows(execute(conn, "SELECT question_id,answer_text FROM nuvedra_assessment_answers WHERE attempt_id=?", (int(active["id"]),)))
            answers = {int(row["question_id"]): str(row.get("answer_text") or "") for row in previous_answers}

                for index, question in enumerate(questions, 1)
            )
            deadline = _deadline_ms(str(active.get("started_at") or ""), settings["time_limit"])

            )

            <form method="post" action="/learn/assessments/{item_id}/attempts/{int(active['id'])}/submit" data-assessment-attempt>
              {question_blocks}
              <button class="studio-button" data-i18n-en="Submit assessment" data-i18n-es="Entregar evaluación">Submit assessment</button>

            assessment_form = f"""
            <form method="post" action="/learn/assessments/{item_id}/start">
              <button class="studio-button" data-i18n-en="Start assessment" data-i18n-es="Comenzar evaluación">Start assessment</button>

        timer_script = """

        (() => {
          const box = document.querySelector('[data-assessment-timer]');
          const form = document.querySelector('[data-assessment-attempt]');
          if (!box || !form) return;
          const clock = box.querySelector('[data-assessment-clock]');

    @app.post("/learn/assessments/{item_id}/start", response_model=None)
    async def start_attempt(item_id: int, request: Request):
        user = academic_access.google_user(request)
        if not user:
            return academic_access.login_redirect(f"/learn/assessments/{item_id}")
        with db() as conn:
            _course_id, item, _module, _access = _student_access(conn, item_id, user)
            questions = _questions(conn, item_id)
            if not questions:
                raise HTTPException(409, "This assessment does not contain questions.")
            settings = _assessment_settings(item)
            existing_active = rows(
                execute(
                    conn,
                    "SELECT id FROM nuvedra_assessment_attempts WHERE item_id=? AND lower(student_email)=? AND status='in_progress'",
                    (item_id, user["email"].lower()),
                )
            )
            if existing_active:
                return RedirectResponse(f"/learn/assessments/{item_id}", status_code=303)
            count = rows(
                execute(
                    conn,
                    "SELECT COUNT(*) AS total FROM nuvedra_assessment_attempts WHERE item_id=? AND lower(student_email)=? AND status<>'in_progress'",
                    (item_id, user["email"].lower()),
                )
            )[0]["total"]
            if int(count or 0) >= settings["attempts"]:
                raise HTTPException(409, "No attempts remain.")
            attempt_number = int(count or 0) + 1
            attempt_id = _insert_id(
                conn,
                """INSERT INTO nuvedra_assessment_attempts
                   (item_id,student_email,attempt_number,status,started_at,score_auto,has_manual_items)
                   VALUES (?,?,?,?,?,?,?)""",
                (item_id, user["email"].lower(), attempt_number, "in_progress", utcnow(), 0, 0),
            )
            audit(conn, user["email"], "assessment_attempt_started", "attempt", str(attempt_id), str(item_id), request.client.host if request.client else "")
        return RedirectResponse(f"/learn/assessments/{item_id}", status_code=303)

    @app.post("/learn/assessments/{item_id}/attempts/{attempt_id}/submit", response_model=None)
    async def submit_attempt(item_id: int, attempt_id: int, request: Request):
        user = academic_access.google_user(request)
        if not user:
            return academic_access.login_redirect(f"/learn/assessments/{item_id}")
        form = await request.form()
        with db() as conn:
            _course_id, item, _module, _access = _student_access(conn, item_id, user)
            attempts = rows(
                execute(
                    conn,
                    """SELECT * FROM nuvedra_assessment_attempts
                       WHERE id=? AND item_id=? AND lower(student_email)=? AND status='in_progress'""",
                    (attempt_id, item_id, user["email"].lower()),
                )
            )
            if not attempts:
                raise HTTPException(409, "This attempt is not active.")
            attempt = attempts[0]
            questions = _questions(conn, item_id)
            if not questions:
                raise HTTPException(409, "This assessment does not contain questions.")
            settings = _assessment_settings(item)
            now = utcnow()
            late = False
            if settings["time_limit"] > 0:
                elapsed = (_parse_iso(now) - _parse_iso(str(attempt.get("started_at") or now))).total_seconds()
                late = elapsed > settings["time_limit"] * 60 + 30

            auto_score = 0.0
            has_manual = False
            for question in questions:
                qid = int(question["id"])
                answer = str(form.get(f"q_{qid}") or "").strip()
                is_correct, points_awarded = _grade_answer(question, answer)
                if is_correct is None:
                    has_manual = True
                else:
                    auto_score += float(points_awarded or 0)
                found = rows(execute(conn, "SELECT id FROM nuvedra_assessment_answers WHERE attempt_id=? AND question_id=?", (attempt_id, qid)))
                params = (
                    answer,
                    is_correct,
                    points_awarded,
                    now,
                )
                if found:
                    execute(
                        conn,
                        "UPDATE nuvedra_assessment_answers SET answer_text=?,is_correct=?,points_awarded=?,updated_at=? WHERE attempt_id=? AND question_id=?",
                        params + (attempt_id, qid),
                    )
                else:
                    execute(
                        conn,
                        """INSERT INTO nuvedra_assessment_answers
                           (attempt_id,question_id,answer_text,is_correct,points_awarded,created_at,updated_at)
                           VALUES (?,?,?,?,?,?,?)""",
                        (attempt_id, qid, answer, is_correct, points_awarded, now, now),
                    )

            total_score = None if has_manual else auto_score
            status = "submitted_late" if late else "submitted"
            execute(
                conn,
                """UPDATE nuvedra_assessment_attempts
                   SET status=?,submitted_at=?,score_auto=?,score_total=?,has_manual_items=?
                   WHERE id=?""",
                (status, now, auto_score, total_score, 1 if has_manual else 0, attempt_id),
            )
            submission_id = _upsert_submission(conn, item_id, user["email"], int(attempt.get("attempt_number") or 1), status)
            _sync_auto_grade(conn, submission_id, auto_score, user["email"], fully_auto=not has_manual)
            audit(
                conn,
                user["email"],
                "structured_assessment_submitted",
                "attempt",
                str(attempt_id),
                f"{status}:{auto_score:g}",
                request.client.host if request.client else "",
            )
        return RedirectResponse(f"/learn/assessments/{item_id}", status_code=303)
