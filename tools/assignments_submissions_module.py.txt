from __future__ import annotations

from datetime import datetime, timezone
from pathlib import PurePath
from typing import Any

from fastapi import FastAPI, File, Form, HTTPException, Request, UploadFile
from fastapi.responses import HTMLResponse, RedirectResponse, Response

import app.academic_access as academic_access
from app.admin_authoring_v6 import safe_url
from app.admin_console import audit, database_url, db, execute, rows, utcnow

STUDIO_PREFIX = "/faculty/studio"
ASSIGNMENT_TYPES = {"assignment", "project", "presentation"}
MAX_FILE_BYTES = 15 * 1024 * 1024
ALLOWED_EXTENSIONS = {".pdf", ".doc", ".docx", ".ppt", ".pptx", ".xls", ".xlsx", ".txt", ".csv", ".png", ".jpg", ".jpeg", ".zip"}


def _pk() -> str:
    return "INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY" if database_url().startswith("postgres") else "INTEGER PRIMARY KEY AUTOINCREMENT"


def _blob_type() -> str:
    return "BYTEA" if database_url().startswith("postgres") else "BLOB"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def _parse_iso(value: Any) -> datetime | None:
    clean = str(value or "").strip()
    if not clean:
        return None
    try:
        parsed = datetime.fromisoformat(clean.replace("Z", "+00:00"))
    except ValueError:
        return None
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=timezone.utc)
    return parsed.astimezone(timezone.utc)


def _format_when(value: Any) -> str:
    parsed = _parse_iso(value)
    if parsed is None:
        return str(value or "")
    return parsed.astimezone().strftime("%b %d, %Y · %I:%M %p")


def ensure_assignments_schema() -> None:
    pk = _pk()
    blob = _blob_type()
    with db() as conn:
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_assignment_files (
            id {pk}, submission_id INTEGER NOT NULL, filename TEXT NOT NULL,
            content_type TEXT, size_bytes INTEGER NOT NULL, file_bytes {blob} NOT NULL,
            uploaded_at TEXT NOT NULL
        )""")
        execute(conn, f"""CREATE TABLE IF NOT EXISTS nuvedra_assignment_attempts (
            id {pk}, submission_id INTEGER NOT NULL, attempt_no INTEGER NOT NULL,
            response_text TEXT, response_url TEXT, file_id INTEGER,
            status TEXT NOT NULL DEFAULT 'submitted', is_late INTEGER NOT NULL DEFAULT 0,
            submitted_at TEXT NOT NULL, UNIQUE(submission_id,attempt_no)
        )""")


def _bundle(conn: Any, item_id: int) -> tuple[int, dict[str, Any], dict[str, Any]]:
    course_id, item, module = academic_access.item_bundle(conn, item_id)
    if str(item.get("item_type") or "") not in ASSIGNMENT_TYPES:
        raise HTTPException(404, "Assignment not found.")
    return course_id, item, module


def _submission(conn: Any, item_id: int, email: str) -> dict[str, Any] | None:
    found = rows(execute(conn, "SELECT * FROM nuvedra_submissions WHERE item_id=? AND lower(student_email)=?", (item_id, email.lower())))
    return found[0] if found else None


def _latest_file(conn: Any, submission_id: int) -> dict[str, Any] | None:
    found = rows(execute(conn, "SELECT id,submission_id,filename,content_type,size_bytes,uploaded_at FROM nuvedra_assignment_files WHERE submission_id=? ORDER BY id DESC", (submission_id,)))
    return found[0] if found else None


def _attempts(conn: Any, submission_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT a.*,f.filename,f.size_bytes FROM nuvedra_assignment_attempts a
        LEFT JOIN nuvedra_assignment_files f ON f.id=a.file_id
        WHERE a.submission_id=? ORDER BY a.attempt_no DESC,a.id DESC""", (submission_id,)))


def _is_late(item: dict[str, Any]) -> bool:
    due = _parse_iso(item.get("due_at"))
    return bool(due and datetime.now(timezone.utc) > due)


def _notify_instructors(conn: Any, course_id: int, student_email: str, item: dict[str, Any]) -> None:
    recipients = rows(execute(conn, """SELECT lower(user_email) AS email FROM nexus_admin_enrollments
        WHERE course_id=? AND status='active' AND course_role IN ('instructor','teaching_assistant','course_builder','facilitator')
        ORDER BY lower(user_email)""", (course_id,)))
    title = f"Assignment submitted: {item.get('title') or 'Assignment'}"
    message = f"{student_email} submitted work for review."
    now = utcnow()
    for row in recipients:
        execute(conn, "INSERT INTO nuvedra_notifications (recipient_email,course_id,title,message,href,kind,read_at,created_at) VALUES (?,?,?,?,?,'assignment',NULL,?)",
                (row["email"], course_id, title, message, f"{STUDIO_PREFIX}/assignments/{int(item['id'])}", now))


def _validate_file(upload: UploadFile, payload: bytes) -> tuple[str, str]:
    filename = PurePath(upload.filename or "").name
    extension = PurePath(filename).suffix.lower()
    if not filename or extension not in ALLOWED_EXTENSIONS:
        raise HTTPException(400, "This file type is not allowed for assignment submissions.")
    if len(payload) > MAX_FILE_BYTES:
        raise HTTPException(413, "The attachment exceeds the 15 MB assignment limit.")
    return filename, str(upload.content_type or "application/octet-stream")


def _upsert_submission(conn: Any, item_id: int, email: str, response_text: str, response_url: str | None, status: str) -> int:
    found = _submission(conn, item_id, email)
    now = utcnow()
    if found:
        submission_id = int(found["id"])
        submitted_at = now if status == "submitted" else (found.get("submitted_at") or now)
        execute(conn, "UPDATE nuvedra_submissions SET response_text=?,response_url=?,status=?,submitted_at=?,updated_at=? WHERE id=?",
                (response_text, response_url, status, submitted_at, now, submission_id))
        return submission_id
    execute(conn, "INSERT INTO nuvedra_submissions (item_id,student_email,response_text,response_url,status,submitted_at,updated_at) VALUES (?,?,?,?,?,?,?)",
            (item_id, email.lower(), response_text, response_url, status, now, now))
    created = _submission(conn, item_id, email)
    if not created:
        raise HTTPException(500, "Assignment submission could not be stored.")
    return int(created["id"])


def register_assignments_submissions(app: FastAPI) -> None:
    ensure_assignments_schema()

    @app.get("/learn/assignments/{item_id}", response_class=HTMLResponse, response_model=None)
    async def student_assignment(item_id: int, request: Request):
        user = _user(request, f"/learn/assignments/{item_id}")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            course_id, item, module = _bundle(conn, item_id)
            access = academic_access.require_course_role(conn, course_id, user["email"], academic_access.STUDENT_ROLES)
            if str(access.get("course_status")) != "active" or str(module.get("status")) != "published" or str(item.get("status")) != "published":
                raise HTTPException(403, "This assignment is not available.")
            submission = _submission(conn, item_id, user["email"])
            history = _attempts(conn, int(submission["id"])) if submission else []
            latest = _latest_file(conn, int(submission["id"])) if submission else None
        due_text = _format_when(item.get("due_at")) or "No due date"
        status_html = ""
        if submission:
            status = str(submission.get("status") or "draft")
            status_html = f'<p class="studio-notice"><strong data-i18n-en="Current status" data-i18n-es="Estado actual">Current status</strong>: {academic_access.esc(status.title())}</p>'
        history_rows = "".join(
            f'''<tr><td>{int(row.get('attempt_no') or 0)}</td><td>{academic_access.esc(_format_when(row.get('submitted_at')))}</td><td>{'Late' if int(row.get('is_late') or 0) else 'On time'}</td><td>{academic_access.esc(row.get('filename') or '—')}</td></tr>'''
            for row in history
        ) or '<tr><td colspan="4" data-i18n-en="No submitted attempts yet." data-i18n-es="Todavía no hay intentos entregados.">No submitted attempts yet.</td></tr>'
        file_html = ""
        if latest:
            file_html = f'<p><a class="studio-button studio-button--quiet" href="/learn/assignment-files/{int(latest["id"])}" data-i18n-en="Download current attachment" data-i18n-es="Descargar archivo actual">Download current attachment</a> <small>{academic_access.esc(latest.get("filename"))}</small></p>'
        if str(access.get("course_role")) == "student":
            existing_text = submission.get("response_text") if submission else ""
            existing_url = submission.get("response_url") if submission else ""
            form = f'''<section class="studio-panel"><h3 data-i18n-en="Submit assignment" data-i18n-es="Entregar asignación">Submit assignment</h3>{status_html}{file_html}<form method="post" action="/learn/assignments/{item_id}/save" enctype="multipart/form-data" class="studio-form"><label><span data-i18n-en="Written response (optional)" data-i18n-es="Respuesta escrita (opcional)">Written response (optional)</span><textarea name="response_text" rows="7">{academic_access.esc(existing_text)}</textarea></label><label><span data-i18n-en="Evidence link (optional)" data-i18n-es="Enlace de evidencia (opcional)">Evidence link (optional)</span><input type="url" name="response_url" value="{academic_access.esc(existing_url, attr=True)}"></label><label><span data-i18n-en="Attachment (optional, max 15 MB)" data-i18n-es="Archivo (opcional, máximo 15 MB)">Attachment (optional, max 15 MB)</span><input type="file" name="attachment" accept=".pdf,.doc,.docx,.ppt,.pptx,.xls,.xlsx,.txt,.csv,.png,.jpg,.jpeg,.zip"></label><div class="studio-actions"><button class="studio-button studio-button--quiet" name="action" value="draft" data-i18n-en="Save draft" data-i18n-es="Guardar borrador">Save draft</button><button class="studio-button" name="action" value="submit" data-i18n-en="Submit assignment" data-i18n-es="Entregar asignación">Submit assignment</button></div></form></section>'''
        else:
            form = '<section class="studio-panel"><p data-i18n-en="Observers can view assignment instructions but cannot submit work." data-i18n-es="Los observadores pueden consultar las instrucciones, pero no entregar trabajos.">Observers can view assignment instructions but cannot submit work.</p></section>'
        body = f'''{_assets()}<main class="studio-shell" data-studio-root data-testid="student-assignment-v2"><nav class="studio-breadcrumbs"><a href="/learn/courses/{course_id}" data-i18n-en="Back to course" data-i18n-es="Volver al curso">Back to course</a></nav><header class="studio-hero studio-hero--module"><div><p class="studio-eyebrow" data-i18n-en="Assignment" data-i18n-es="Asignación">Assignment</p><h2>{academic_access.esc(item.get('title'))}</h2><p>{academic_access.esc(module.get('title'))} · <span data-i18n-en="Due" data-i18n-es="Entrega">Due</span>: {academic_access.esc(due_text)} · {academic_access.esc(item.get('points') or '—')} pts</p></div></header><article class="studio-panel content-body">{item.get('body_html') or ''}</article>{form}<section class="studio-panel"><h3 data-i18n-en="Submission history" data-i18n-es="Historial de entregas">Submission history</h3><div style="overflow:auto"><table><thead><tr><th data-i18n-en="Attempt" data-i18n-es="Intento">Attempt</th><th data-i18n-en="Submitted" data-i18n-es="Entregado">Submitted</th><th data-i18n-en="Timing" data-i18n-es="Tiempo">Timing</th><th data-i18n-en="Attachment" data-i18n-es="Archivo">Attachment</th></tr></thead><tbody>{history_rows}</tbody></table></div></section></main>'''
        return academic_access.portal_page("Assignment", body, user)

    @app.post("/learn/assignments/{item_id}/save", response_model=None)
    async def save_assignment(item_id: int, request: Request, response_text: str = Form(""), response_url: str = Form(""), action: str = Form("submit"), attachment: UploadFile | None = File(None)):
        user = _user(request, f"/learn/assignments/{item_id}")
        if isinstance(user, RedirectResponse):
            return user
        status = "draft" if action == "draft" else "submitted"
        response = response_text.strip()
        evidence = safe_url(response_url) or None
        payload = b""
        filename = ""
        content_type = ""
        if attachment is not None and attachment.filename:
            payload = await attachment.read(MAX_FILE_BYTES + 1)
            filename, content_type = _validate_file(attachment, payload)
        if status == "submitted" and not response and not evidence and not payload:
            raise HTTPException(400, "Add a written response, evidence link, or attachment before submitting.")
        with db() as conn:
            course_id, item, module = _bundle(conn, item_id)
            access = academic_access.require_course_role(conn, course_id, user["email"], {"student"})
            if str(access.get("course_status")) != "active" or str(module.get("status")) != "published" or str(item.get("status")) != "published":
                raise HTTPException(403, "This assignment is not available.")
            submission_id = _upsert_submission(conn, item_id, user["email"], response, evidence, status)
            file_id = None
            if payload:
                now = utcnow()
                execute(conn, "INSERT INTO nuvedra_assignment_files (submission_id,filename,content_type,size_bytes,file_bytes,uploaded_at) VALUES (?,?,?,?,?,?)",
                        (submission_id, filename, content_type, len(payload), payload, now))
                found_file = rows(execute(conn, "SELECT id FROM nuvedra_assignment_files WHERE submission_id=? ORDER BY id DESC", (submission_id,)))
                file_id = int(found_file[0]["id"])
            if status == "submitted":
                attempts = rows(execute(conn, "SELECT COUNT(*) AS total FROM nuvedra_assignment_attempts WHERE submission_id=?", (submission_id,)))
                attempt_no = int(attempts[0].get("total") or 0) + 1
                now = utcnow()
                execute(conn, "INSERT INTO nuvedra_assignment_attempts (submission_id,attempt_no,response_text,response_url,file_id,status,is_late,submitted_at) VALUES (?,?,?,?,?,'submitted',?,?)",
                        (submission_id, attempt_no, response, evidence, file_id, 1 if _is_late(item) else 0, now))
                _notify_instructors(conn, course_id, user["email"], item)
                audit(conn, user["email"], "assignment_submitted", "item", str(item_id), f"attempt={attempt_no}", request.client.host if request.client else "")
            else:
                audit(conn, user["email"], "assignment_draft_saved", "item", str(item_id), "draft", request.client.host if request.client else "")
        return RedirectResponse(f"/learn/assignments/{item_id}", status_code=303)

    @app.get("/learn/assignment-files/{file_id}", response_model=None)
    async def download_assignment_file(file_id: int, request: Request):
        user = _user(request, "/portal")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            found = rows(execute(conn, """SELECT f.*,s.student_email,i.id AS item_id,m.course_id
                FROM nuvedra_assignment_files f
                JOIN nuvedra_submissions s ON s.id=f.submission_id
                JOIN nexus_content_items i ON i.id=s.item_id
                JOIN nexus_modules m ON m.id=i.module_id
                WHERE f.id=?""", (file_id,)))
            if not found:
                raise HTTPException(404, "Assignment attachment not found.")
            file_row = found[0]
            owner = str(file_row.get("student_email") or "").lower() == user["email"].lower()
            if not owner:
                academic_access.require_course_role(conn, int(file_row["course_id"]), user["email"], academic_access.AUTHOR_ROLES)
            payload = file_row.get("file_bytes")
        headers = {"Content-Disposition": f'attachment; filename="{PurePath(str(file_row.get("filename") or "assignment-file")).name}"'}
        return Response(content=bytes(payload or b""), media_type=str(file_row.get("content_type") or "application/octet-stream"), headers=headers)

        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/assignments")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            course = academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            data = rows(execute(conn, """SELECT i.id,i.title,i.item_type,i.points,i.due_at,i.status,m.title AS module_title,
                (SELECT COUNT(*) FROM nuvedra_submissions s WHERE s.item_id=i.id AND s.status='submitted') AS submitted_count

        body = f'''{_assets()}<main class="studio-shell" data-studio-root data-testid="course-assignments-v2"><nav class="studio-breadcrumbs"><a href="{STUDIO_PREFIX}/courses/{course_id}" data-i18n-en="Back to Course Studio" data-i18n-es="Volver a Course Studio">Back to Course Studio</a></nav><header class="studio-hero"><div><p class="studio-eyebrow" data-i18n-en="Assignments & Submissions v2" data-i18n-es="Asignaciones y entregas v2">Assignments & Submissions v2</p><h2>{academic_access.esc(course.get('course_code'))}: {academic_access.esc(course.get('title'))}</h2><p data-i18n-en="Review assignment activity, submission counts, due dates, and student evidence." data-i18n-es="Revise actividad de asignaciones, entregas, fechas límite y evidencia estudiantil.">Review assignment activity, submission counts, due dates, and student evidence.</p></div><div class="studio-actions"><a class="studio-button studio-button--quiet" href="{STUDIO_PREFIX}/courses/{course_id}/gradebook" data-i18n-en="Gradebook" data-i18n-es="Calificaciones">Gradebook</a></div></header><section class="studio-grid">{cards}</section></main>'''

    @app.get(f"{STUDIO_PREFIX}/assignments/{{item_id}}", response_class=HTMLResponse, response_model=None)
    async def instructor_assignment_inbox(item_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/assignments/{item_id}")
        if isinstance(user, RedirectResponse):
            return user
        with db() as conn:
            course_id, item, module = _bundle(conn, item_id)
            course = academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            submissions = rows(execute(conn, """SELECT s.*,g.points_awarded,g.feedback,
                (SELECT id FROM nuvedra_assignment_files f WHERE f.submission_id=s.id ORDER BY f.id DESC LIMIT 1) AS file_id,
                (SELECT filename FROM nuvedra_assignment_files f WHERE f.submission_id=s.id ORDER BY f.id DESC LIMIT 1) AS filename,
                (SELECT COUNT(*) FROM nuvedra_assignment_attempts a WHERE a.submission_id=s.id) AS attempt_count,
                (SELECT is_late FROM nuvedra_assignment_attempts a WHERE a.submission_id=s.id ORDER BY a.attempt_no DESC LIMIT 1) AS latest_late
                FROM nuvedra_submissions s LEFT JOIN nuvedra_grades g ON g.submission_id=s.id
                WHERE s.item_id=? ORDER BY lower(s.student_email)""", (item_id,)))
        cards = []
        for row in submissions:
            file_link = f'<a href="/learn/assignment-files/{int(row["file_id"])}" data-i18n-en="Download attachment" data-i18n-es="Descargar archivo">Download attachment</a> · {academic_access.esc(row.get("filename"))}' if row.get("file_id") else '—'
            grade = "Pending" if row.get("points_awarded") is None else f"{float(row.get('points_awarded') or 0):g} / {float(item.get('points') or 0):g}"
            cards.append(f'''<article class="studio-panel"><p class="studio-eyebrow">{academic_access.esc(row.get('student_email'))}</p><h3>{academic_access.esc(row.get('status'))} · {int(row.get('attempt_count') or 0)} attempt(s)</h3><p><strong>{'Late' if int(row.get('latest_late') or 0) else 'On time'}</strong> · {academic_access.esc(_format_when(row.get('submitted_at')))}</p><details><summary data-i18n-en="View written response" data-i18n-es="Ver respuesta escrita">View written response</summary><p>{academic_access.esc(row.get('response_text') or '—')}</p><p>{academic_access.esc(row.get('response_url') or '')}</p></details><p>{file_link}</p><p><strong data-i18n-en="Grade" data-i18n-es="Calificación">Grade</strong>: {academic_access.esc(grade)}</p></article>''')
        content = "".join(cards) or '<section class="studio-empty" data-i18n-en="No student submissions yet." data-i18n-es="Todavía no hay entregas estudiantiles.">No student submissions yet.</section>'
        body = f'''{_assets()}<main class="studio-shell" data-studio-root data-testid="assignment-submission-inbox-v2"><nav class="studio-breadcrumbs"><a href="{STUDIO_PREFIX}/courses/{course_id}/assignments" data-i18n-en="Assignments" data-i18n-es="Asignaciones">Assignments</a></nav><header class="studio-hero studio-hero--module"><div><p class="studio-eyebrow">{academic_access.esc(module.get('title'))}</p><h2>{academic_access.esc(item.get('title'))}</h2><p>{academic_access.esc(course.get('course_code'))} · {academic_access.esc(item.get('points') or '—')} pts · {academic_access.esc(_format_when(item.get('due_at')) or 'No due date')}</p></div><div class="studio-actions"><a class="studio-button" href="{STUDIO_PREFIX}/courses/{course_id}/gradebook" data-i18n-en="Open Gradebook" data-i18n-es="Abrir calificaciones">Open Gradebook</a></div></header><section class="studio-grid">{content}</section></main>'''
