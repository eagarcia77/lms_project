STUDIO_PREFIX = "/faculty/studio"


def _assets() -> str:
    return '<link rel="stylesheet" href="/static/course-studio.css"><script src="/static/course-studio.js" defer></script>'


def _author(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)
    if not user:
        return academic_access.login_redirect(return_to)
    return user


def _parse_iso(value: Any) -> datetime | None:
    clean = str(value or "").strip()
    if not clean:
        return None
    try:
        parsed = datetime.fromisoformat(clean.replace("Z", "+00:00"))
    except ValueError:
        return None
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=timezone.utc)
    return parsed.astimezone(timezone.utc)


def _published_items(conn: Any, course_id: int) -> list[dict[str, Any]]:
    return rows(execute(conn, """SELECT i.id,i.item_type,i.title,i.points,i.due_at,i.position,
        m.id AS module_id,m.title AS module_title,m.position AS module_position
        FROM nexus_content_items i JOIN nexus_modules m ON m.id=i.module_id
        WHERE m.course_id=? AND m.status='published' AND i.status='published'
        ORDER BY m.position,m.id,i.position,i.id""", (course_id,)))


def _completion_map(conn: Any, item_ids: list[int], email: str) -> dict[int, bool]:
    result = {item_id: False for item_id in item_ids}
    if not item_ids:
        return result
    placeholders = ",".join("?" for _ in item_ids)
    params = tuple(item_ids) + (email.lower(),)
    for row in rows(execute(conn, f"SELECT item_id,status FROM nuvedra_content_progress WHERE item_id IN ({placeholders}) AND lower(student_email)=?", params)):
        result[int(row["item_id"])] = str(row.get("status")) == "completed"
    for row in rows(execute(conn, f"SELECT item_id FROM nuvedra_submissions WHERE item_id IN ({placeholders}) AND lower(student_email)=? AND status='submitted'", params)):
        result[int(row["item_id"])] = True
    for row in rows(execute(conn, f"SELECT item_id FROM nuvedra_assessment_attempts WHERE item_id IN ({placeholders}) AND lower(student_email)=? AND status<>'in_progress'", params)):
        result[int(row["item_id"])] = True
    return result


def _grade_summary(conn: Any, course_id: int, email: str) -> tuple[float | None, int]:
    data = rows(execute(conn, """SELECT g.points_awarded,i.points
        FROM nuvedra_grades g
        JOIN nuvedra_submissions s ON s.id=g.submission_id
        JOIN nexus_content_items i ON i.id=s.item_id
        JOIN nexus_modules m ON m.id=i.module_id
        WHERE m.course_id=? AND lower(s.student_email)=? AND g.points_awarded IS NOT NULL
        ORDER BY g.id""", (course_id, email.lower())))
    earned = 0.0
    possible = 0.0
    graded = 0
    for row in data:
        points = row.get("points")
        if points in (None, "") or float(points or 0) <= 0:
            continue
        earned += float(row.get("points_awarded") or 0)
        possible += float(points)
        graded += 1
    return ((earned / possible) * 100 if possible > 0 else None), graded


def _submission_count(conn: Any, course_id: int, email: str) -> int:
    legacy = rows(execute(conn, """SELECT COUNT(*) AS total FROM nuvedra_submissions s
        JOIN nexus_content_items i ON i.id=s.item_id JOIN nexus_modules m ON m.id=i.module_id
        WHERE m.course_id=? AND lower(s.student_email)=? AND s.status='submitted'""", (course_id, email.lower())))
    structured = rows(execute(conn, """SELECT COUNT(*) AS total FROM nuvedra_assessment_attempts a
        JOIN nexus_content_items i ON i.id=a.item_id JOIN nexus_modules m ON m.id=i.module_id
        WHERE m.course_id=? AND lower(a.student_email)=? AND a.status<>'in_progress'""", (course_id, email.lower())))
    return int(legacy[0].get("total") or 0 if legacy else 0) + int(structured[0].get("total") or 0 if structured else 0)


def _student_metrics(conn: Any, course_id: int) -> tuple[list[dict[str, Any]], int]:
    students = rows(execute(conn, """SELECT lower(user_email) AS student_email FROM nexus_admin_enrollments
        WHERE course_id=? AND status='active' AND course_role='student' ORDER BY lower(user_email)""", (course_id,)))
    items = _published_items(conn, course_id)
    item_ids = [int(item["id"]) for item in items]
    now = datetime.now(timezone.utc)
    metrics: list[dict[str, Any]] = []
    total_overdue = 0
    for student in students:
        email = str(student.get("student_email") or "").lower()
        completion = _completion_map(conn, item_ids, email)
        completed = sum(1 for item_id in item_ids if completion.get(item_id, False))
        total = len(item_ids)
        progress = round((completed / total) * 100) if total else 0
        overdue = 0
        for item in items:
            due = _parse_iso(item.get("due_at"))
            if due is not None and due < now and not completion.get(int(item["id"]), False):
                overdue += 1
        total_overdue += overdue
        grade_percent, graded = _grade_summary(conn, course_id, email)
        submissions = _submission_count(conn, course_id, email)
        if overdue > 0 or progress < 50:
            attention = "needs-attention"
            label_en, label_es = "Needs attention", "Requiere atención"
        elif progress < 75:
            attention = "watch"
            label_en, label_es = "Watch", "Observar"
        else:
            attention = "on-track"
            label_en, label_es = "On track", "En progreso adecuado"
        metrics.append({
            "email": email,
            "completed": completed,
            "total": total,
            "progress": progress,
            "overdue": overdue,
            "submissions": submissions,
            "grade_percent": grade_percent,
            "graded": graded,
            "attention": attention,
            "label_en": label_en,
            "label_es": label_es,
        })
    return metrics, total_overdue
