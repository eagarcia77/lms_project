def _user(request: Request, return_to: str) -> dict[str, Any] | RedirectResponse:
    user = academic_access.google_user(request)

def _admin(request: Request) -> dict[str, Any]:
    user = admin_console.session_user(request)
    if not user or str(user.get("role") or "") not in {"superadmin", "course_admin"}:
        raise HTTPException(403, "Platform administration permission is required.")
    return user


def _connection_scopes(conn: Any, email: str) -> set[str]:
    connection = m365._connection(conn, email)
    if not connection:
        return set()
    return {x.strip() for x in str(connection.get("scopes") or "").split() if x.strip()}


def _has_scope(scopes: set[str], *options: str) -> bool:
    return any(option in scopes for option in options)


def _require_scope(conn: Any, email: str, *options: str) -> set[str]:
    scopes = _connection_scopes(conn, email)
    if not _has_scope(scopes, *options):
        raise HTTPException(403, "The connected Microsoft account has not granted the required delegated Graph permission: " + " or ".join(options))
    return scopes


def _safe_graph_target(target: str) -> str:
    clean = str(target or "").strip()
    if clean.startswith("/"):
        return m365.GRAPH_ROOT + clean
    parsed = urlparse(clean)
    if parsed.scheme != "https" or parsed.hostname != "graph.microsoft.com" or not parsed.path.startswith("/v1.0/"):
        raise HTTPException(502, "Microsoft Graph returned an unsafe continuation URL.")
    return clean


async def _graph_raw(email: str, method: str, target: str, *, params: dict[str, Any] | None = None, json_body: dict[str, Any] | None = None) -> tuple[int, dict[str, str], dict[str, Any]]:
    url = _safe_graph_target(target)
    token = await m365._access_token(email)
    response: httpx.Response | None = None
    for attempt in range(2):
        try:
            async with httpx.AsyncClient(timeout=20.0, follow_redirects=False) as client:
                response = await client.request(method.upper(), url, params=params, json=json_body, headers={"Authorization": f"Bearer {token}", "Accept": "application/json"})
        except httpx.HTTPError as exc:
            raise HTTPException(502, "Microsoft Graph is unavailable.") from exc
        if response.status_code != 401 or attempt == 1:
            break
        token = await m365._access_token(email, force_refresh=True)
    assert response is not None
    try:
        body = response.json() if response.content else {}
    except ValueError:
        body = {}
    if response.status_code >= 400:
        error = body.get("error") if isinstance(body, dict) else None
        message = error.get("message") if isinstance(error, dict) else None
        status = response.status_code if response.status_code in {400, 401, 403, 404, 409, 429} else 502
        raise HTTPException(status, str(message or "Microsoft Graph request failed.")[:700])
    headers = {str(k).lower(): str(v) for k, v in response.headers.items()}
    return response.status_code, headers, body if isinstance(body, dict) else {}


async def _graph_collection(email: str, path: str, *, params: dict[str, Any] | None = None) -> tuple[list[dict[str, Any]], int]:
    target = path
    first_params = params
    result: list[dict[str, Any]] = []
    pages = 0
    while target:
        pages += 1
        if pages > MAX_GRAPH_PAGES:
            raise HTTPException(409, "Microsoft Graph pagination exceeded the NUVEDRA safety limit.")
        _status, _headers, payload = await _graph_raw(email, "GET", target, params=first_params)
        first_params = None
        values = payload.get("value", [])
        if isinstance(values, list):
            result.extend(x for x in values if isinstance(x, dict))
        if len(result) > MAX_GRAPH_ITEMS:
            raise HTTPException(409, "Microsoft Graph returned more members than the NUVEDRA safety limit permits.")
        next_link = str(payload.get("@odata.nextLink") or "").strip()
        target = next_link if next_link else ""
    return result, pages


def _team_id_from_location(value: str) -> str:
    match = TEAM_LOCATION_RE.search(str(value or ""))
    return match.group(1) if match else ""


def _roster_hash(conn: Any, course_id: int) -> str:
    rows_data = education._course_members(conn, course_id)
    canonical = "\n".join(sorted(f"{str(x['email']).lower()}|{x['course_role']}" for x in rows_data))
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


async def _membership_plan(conn: Any, actor_email: str, course_id: int, team: dict[str, Any]) -> dict[str, Any]:
    target = education._course_members(conn, course_id)
    current, pages = await _graph_collection(actor_email, f"/teams/{education._team_id(str(team['team_id']))}/members", params={"$top": 200})
    by_email = {education._member_email(member): member for member in current if education._member_email(member)}
    target_emails = {str(x["email"]).lower() for x in target}
    missing: list[dict[str, Any]] = []
    matched: list[dict[str, Any]] = []
    role_mismatches: list[dict[str, Any]] = []
    for row in target:
        email = str(row["email"]).lower()
        role = str(row["course_role"])
        member = by_email.get(email)
        if not member:
            missing.append({"email": email, "course_role": role, "teams_role": "owner" if education._target_owner(role) else "member"})
            continue
        roles = {str(x).lower() for x in (member.get("roles") or [])}
        matched.append({"email": email, "course_role": role, "teams_roles": sorted(roles), "microsoft_user_id": str(member.get("userId") or member.get("id") or "")})
        if education._target_owner(role) and "owner" not in roles:
            role_mismatches.append({"email": email, "expected": "owner", "actual": sorted(roles) or ["member"]})
    unmanaged = [{"email": education._member_email(member) or "(no email exposed)", "display_name": member.get("displayName")} for member in current if not education._member_email(member) or education._member_email(member) not in target_emails]
    return {"target_count": len(target), "team_member_count": len(current), "pages": pages, "missing": missing, "matched": matched, "role_mismatches": role_mismatches, "unmanaged": unmanaged, "sync_mode": "incremental-additive", "roster_hash": _roster_hash(conn, course_id)}


def _save_roster_state(conn: Any, course_id: int, email: str, role: str, microsoft_user_id: str, status: str, action: str) -> None:
    execute(conn, """INSERT INTO nuvedra_microsoft_roster_state
        (course_id,user_email,course_role,microsoft_user_id,status,last_action,updated_at)
        VALUES (?,?,?,?,?,?,?)
        ON CONFLICT(course_id,user_email) DO UPDATE SET course_role=excluded.course_role,
        microsoft_user_id=excluded.microsoft_user_id,status=excluded.status,
        last_action=excluded.last_action,updated_at=excluded.updated_at""",
        (course_id, email.lower(), role, microsoft_user_id or None, status, action, utcnow()))


async def _permission_probe(email: str, course_id: int | None, team_id: str = "") -> dict[str, Any]:
    checks: list[tuple[str, str, dict[str, Any] | None]] = [
        ("User.Read", "/me", {"$select": "id,userPrincipalName"}),
        ("Team.ReadBasic.All", "/me/joinedTeams", {"$top": 1, "$select": "id,displayName"}),
        ("User.ReadBasic.All", "/users", {"$top": 1, "$select": "id,userPrincipalName"}),
    ]
    if team_id:
        clean = education._team_id(team_id)
        checks.extend([
            ("TeamMember.ReadWrite.All", f"/teams/{clean}/members", {"$top": 1}),
            ("Channel.ReadBasic.All", f"/teams/{clean}/channels", {"$top": 1, "$select": "id,displayName"}),
            ("Files.Read", f"/groups/{clean}/drive/root/children", {"$top": 1, "$select": "id,name,webUrl"}),
        ])
    results: dict[str, Any] = {}
    for label, path, params in checks:
        try:
            _status, _headers, payload = await _graph_raw(email, "GET", path, params=params)
            results[label] = {"ok": True, "count": len(payload.get("value", [])) if isinstance(payload.get("value"), list) else 1}
        except HTTPException as exc:
            results[label] = {"ok": False, "status": exc.status_code, "detail": str(exc.detail)[:220]}
    return {"course_id": course_id, "checks": results, "probed_at": utcnow()}


def _course_instructor(conn: Any, course_id: int, email: str) -> None:
    found = rows(execute(conn, "SELECT course_role FROM nexus_admin_enrollments WHERE course_id=? AND lower(user_email)=? AND status='active'", (course_id, email.lower())))
    if not found or str(found[0].get("course_role") or "") != "instructor":
        raise HTTPException(403, "Controlled Microsoft Team creation is limited to the assigned NUVEDRA instructor.")


def _upsert_team(conn: Any, course_id: int, team_id: str, name: str, web_url: str, actor: str) -> None:
    now = utcnow()
    existing = rows(execute(conn, "SELECT id FROM nuvedra_microsoft_course_teams WHERE course_id=?", (course_id,)))
    if existing:
        execute(conn, "UPDATE nuvedra_microsoft_course_teams SET team_id=?,display_name=?,web_url=?,status='active',linked_by=?,linked_at=? WHERE course_id=?", (team_id, name, web_url or None, actor, now, course_id))
    else:
        execute(conn, "INSERT INTO nuvedra_microsoft_course_teams (course_id,team_id,display_name,web_url,status,linked_by,linked_at,last_synced_at,last_sync_json) VALUES (?,?,?,?,'active',?,?,NULL,NULL)", (course_id, team_id, name, web_url or None, actor, now))

        return academic_access.portal_page("Microsoft 365 production roster", body, user)

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/microsoft365/production/sync", response_model=None)
    async def production_sync(course_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/production")
        if isinstance(user, RedirectResponse): return user
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            _require_scope(conn, user["email"], "TeamMember.ReadWrite.All")
            team = education._team_link(conn, course_id)
            if not team: raise HTTPException(409, "Link or provision a Microsoft Team first.")
            plan = await _membership_plan(conn, user["email"], course_id, team)
            prior = {str(x["user_email"]).lower(): x for x in rows(execute(conn, "SELECT * FROM nuvedra_microsoft_roster_state WHERE course_id=?", (course_id,)))}
        added: list[str] = []
        for item in plan["missing"]:
            target_email = str(item["email"]).lower()
            cached_id = str((prior.get(target_email) or {}).get("microsoft_user_id") or "")
            graph_user = {"id": cached_id} if cached_id else await education._resolve_graph_user(user["email"], target_email)
            user_id = str(graph_user.get("id") or "")
            if not user_id: raise HTTPException(409, f"Microsoft Entra user could not be resolved for {target_email}.")
            await m365._graph_json(user["email"], "POST", f"/teams/{education._team_id(str(team['team_id']))}/members", json_body=education._member_payload(user_id, education._target_owner(str(item["course_role"]))))
            added.append(target_email)
            with db() as conn: _save_roster_state(conn, course_id, target_email, str(item["course_role"]), user_id, "present", "added")
        with db() as conn:
            current_targets = {str(x["email"]).lower(): x for x in education._course_members(conn, course_id)}
            for item in plan["matched"]:
                _save_roster_state(conn, course_id, str(item["email"]), str(item["course_role"]), str(item.get("microsoft_user_id") or ""), "present", "unchanged")
            for email, state in prior.items():
                if email not in current_targets:
                    _save_roster_state(conn, course_id, email, str(state.get("course_role") or "student"), str(state.get("microsoft_user_id") or ""), "unenrolled_preserved", "preserved_in_teams")
            summary = {"mode": "incremental-additive", "added": added, "added_count": len(added), "unchanged_count": len(plan["matched"]), "external_preserved_count": len(plan["unmanaged"]), "pages": plan["pages"], "roster_hash": plan["roster_hash"]}
            education._record_sync(conn, course_id, str(team["team_id"]), user["email"], "production_incremental_sync", summary)
            audit(conn, user["email"], "microsoft365_production_incremental_sync", "course", str(course_id), json.dumps(summary), request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/production", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/microsoft365/production/team", response_model=None)
    async def create_course_team(course_id: int, request: Request, display_name: str = Form(...)):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/production")
        if isinstance(user, RedirectResponse): return user
        if not _truthy("MICROSOFT_ALLOW_TEAM_CREATION"):
            raise HTTPException(403, "Controlled Microsoft Team creation is disabled by platform policy.")
        clean_name = display_name.strip()
        if not clean_name or len(clean_name) > 120: raise HTTPException(400, "Team display name must contain 1–120 characters.")
        with db() as conn:
            course = academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            _course_instructor(conn, course_id, user["email"])
            _require_scope(conn, user["email"], "Team.Create")
            if education._team_link(conn, course_id): raise HTTPException(409, "This NUVEDRA course already has a linked Microsoft Team.")
        payload = {"template@odata.bind": "https://graph.microsoft.com/v1.0/teamsTemplates('standard')", "displayName": clean_name, "description": str(course.get("description") or f"NUVEDRA course {course.get('course_code')}")[:1000], "visibility": "Private", "firstChannelName": "General"}
        status, headers, body = await _graph_raw(user["email"], "POST", "/teams", json_body=payload)
        if status not in {201, 202}: raise HTTPException(502, "Microsoft Graph did not accept the Team creation request.")
        location = headers.get("content-location") or headers.get("location") or str(body.get("targetResourceLocation") or "")
        operation = headers.get("location") or str(body.get("operationUrl") or "")
        team_id = _team_id_from_location(location) or str(body.get("targetResourceId") or "")
        now = utcnow()
        with db() as conn:
            execute(conn, """INSERT INTO nuvedra_microsoft_team_provisioning
                (course_id,requested_by,display_name,team_id,operation_url,status,details_json,requested_at,checked_at)
                VALUES (?,?,?,?,?,'accepted',?,?,NULL)
                ON CONFLICT(course_id) DO UPDATE SET requested_by=excluded.requested_by,display_name=excluded.display_name,
                team_id=excluded.team_id,operation_url=excluded.operation_url,status='accepted',details_json=excluded.details_json,
                requested_at=excluded.requested_at,checked_at=NULL""", (course_id, user["email"], clean_name, team_id or None, operation or None, json.dumps({"http_status": status}), now))
            audit(conn, user["email"], "microsoft365_team_creation_requested", "course", str(course_id), clean_name, request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/production", status_code=303)

    @app.post(f"{STUDIO_PREFIX}/courses/{{course_id}}/microsoft365/production/team/check", response_model=None)
    async def check_course_team(course_id: int, request: Request):
        user = _user(request, f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/production")
        if isinstance(user, RedirectResponse): return user
        with db() as conn:
            academic_access.require_course_role(conn, course_id, user["email"], academic_access.AUTHOR_ROLES)
            found = rows(execute(conn, "SELECT * FROM nuvedra_microsoft_team_provisioning WHERE course_id=?", (course_id,)))
        if not found: raise HTTPException(404, "No Team provisioning request exists for this course.")
        provision = found[0]
        team_id = str(provision.get("team_id") or "")
        if not team_id and provision.get("operation_url"):
            _status, _headers, op = await _graph_raw(user["email"], "GET", str(provision["operation_url"]))
            team_id = str(op.get("targetResourceId") or "") or _team_id_from_location(str(op.get("targetResourceLocation") or ""))
        if not team_id:
            with db() as conn: execute(conn, "UPDATE nuvedra_microsoft_team_provisioning SET status='provisioning',checked_at=? WHERE course_id=?", (utcnow(), course_id))
            raise HTTPException(409, "Microsoft accepted the Team request, but provisioning is still in progress.")
        try:
            team = await m365._graph_json(user["email"], "GET", f"/teams/{education._team_id(team_id)}", params={"$select": "id,displayName,webUrl"})
        except HTTPException as exc:
            if exc.status_code == 404:
                with db() as conn: execute(conn, "UPDATE nuvedra_microsoft_team_provisioning SET team_id=?,status='provisioning',checked_at=? WHERE course_id=?", (team_id, utcnow(), course_id))
                raise HTTPException(409, "The Team identifier exists but Microsoft is still provisioning the Team.") from exc
            raise
        with db() as conn:
            _upsert_team(conn, course_id, team_id, str(team.get("displayName") or provision.get("display_name") or "Microsoft Team"), str(team.get("webUrl") or ""), user["email"])
            execute(conn, "UPDATE nuvedra_microsoft_team_provisioning SET team_id=?,status='ready',checked_at=?,details_json=? WHERE course_id=?", (team_id, utcnow(), json.dumps({"verified": True}), course_id))
            audit(conn, user["email"], "microsoft365_team_provisioned", "course", str(course_id), team_id, request.client.host if request.client else "")
        return RedirectResponse(f"{STUDIO_PREFIX}/courses/{course_id}/microsoft365/production", status_code=303)
